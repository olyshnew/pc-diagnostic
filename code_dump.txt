// getFullReport собирает полный текстовый отчёт и структурированный Report
// для сохранения в JSON/CSV/HTML. Возвращает обе версии.
func getFullReport(hwResults []hardware.DiagnosticResult) (string, report.Report) {
	var reportText strings.Builder
	reportText.WriteString("ОТЧЁТ ДИАГНОСТИКИ ПК\n")
	reportText.WriteString(fmt.Sprintf("Дата: %s\n", time.Now().Format("2006-01-02 15:04:05")))
	reportText.WriteString(fmt.Sprintf("ОС: %s (%s)\n\n", runtime.GOOS, runtime.GOARCH))

	// Структурированный отчёт: секции из результатов оборудования
	structured := report.BuildFromHardware(hwResults)

	for _, res := range hwResults {
		reportText.WriteString(res.Name + ":\n")
		if res.Error != nil {
			reportText.WriteString(fmt.Sprintf(" Ошибка: %v\n", res.Error))
		} else {
			val := stripFirstLine(res.Value)
			if len(val) > 2000 {
				val = val[:2000] + "\n... [Вывод обрезан]"
			}
			reportText.WriteString(val)
		}
		reportText.WriteString("\n---\n\n")
	}

	reportText.WriteString("=== РАСШИРЕННАЯ ИНФОРМАЦИЯ ===\n\n")

	utils.Println(utils.Info("[•] Сбор системной информации (драйверы, автозагрузка, целостность...)..."))

	// Параллельный сбор системной информации.
	sysFuncs := []struct {
		Title string
		Fn    func() string
	}{
		{"Вентиляторы и напряжения", hardware.GetSensorsInfo},
		{"Диски (WMI)", system.GetDiskInfo},
		{"Материнская плата (WMI)", system.GetMainboardInfo},
		{"Драйверы (WMI)", system.GetDriversInfo},
		{"USB-устройства (WMI)", system.GetUSBInfo},
		{"Автозагрузка", system.GetAutostartInfo},
		{"Процессы", system.GetProcessesInfo},
		{"Службы", system.GetServicesInfo},
		{"Обновления ОС", system.GetUpdatesInfo},
		{"Целостность системы", system.CheckSystemIntegrity},
		{"Скорость интернета", system.TestInternetSpeed},
		{"Звуковые устройства", system.GetAudioInfo},

		// Диски, расширенная диагностика
		{"Проверка файловой системы", hardware.CheckFileSystemInfo},
		{"Износ SSD (TBW)", hardware.GetSSDWearInfo},
		{"Фрагментация", hardware.AnalyzeFragmentation},
		{"Случайный доступ 4K", hardware.BenchmarkDiskRandom},

		// Сеть, расширенная диагностика
		{"Traceroute", system.GetTracerouteInfo},
		{"Открытые порты", system.ScanOpenPorts},
		{"Активные соединения", system.GetActiveConnectionsInfo},
		{"DNS-диагностика", system.GetDNSInfo},
	}

	sysResults := make([]string, len(sysFuncs))
	var wg sync.WaitGroup

	for i, sf := range sysFuncs {
		wg.Add(1)
		go func(idx int, f func() string) {
			defer wg.Done()
			sysResults[idx] = f()
		}(i, sf.Fn)
	}

	wg.Wait()
	utils.Println(utils.Success("[✓] Системная информация собрана."))

	for i, text := range sysResults {
		reportText.WriteString(text + "\n")
		structured.AddText(sysFuncs[i].Title, text)
	}

	utils.Println(utils.Info("[•] Проверка дисков (SMART)..."))
	smartPath, err := exec.LookPath("smartctl")
	if err == nil && smartPath != "" {
		deepResults := hardware.RunSMARTDiagnostics()
		for _, res := range deepResults {
			reportText.WriteString(fmt.Sprintf("%s:\n", res.Name))
			if res.Error != nil {
				reportText.WriteString(fmt.Sprintf(" [ОШИБКА - от Администратора]: %v\n", res.Error))
				structured.AddText("S.M.A.R.T. — "+res.Name, "ОШИБКА: "+res.Error.Error())
			} else {
				clean := strings.TrimSpace(res.Value)
				reportText.WriteString(clean + "\n")
				structured.AddText("S.M.A.R.T. — "+res.Name, clean)
			}
			reportText.WriteString("\n")
		}

		utils.Println(utils.Info("[•] Тест скорости диска..."))
		wSpd, rSpd, benchErr := hardware.BenchmarkDisk(100)
		if benchErr == nil {
			benchLine := fmt.Sprintf("Запись: %.2f МБ/с | Чтение: %.2f МБ/с", wSpd, rSpd)
			reportText.WriteString(fmt.Sprintf("Тест скорости диска:\n %s\n", benchLine))
			structured.AddText("Тест скорости диска", benchLine)
		}
	} else {
		smartMsg := "Утилита 'smartctl' не найдена.\nСкачайте Smartmontools."
		reportText.WriteString("S.M.A.R.T.: " + smartMsg + "\n")
		structured.AddText("S.M.A.R.T. диагностика дисков", smartMsg)
	}

	utils.Println(utils.Success("[✓] Отчёт сформирован."))
	return reportText.String(), structured
}

func saveReportToFile(filename, reportText string) error {
	return os.WriteFile(filename, []byte(reportText), 0644)
}

// saveReportAnyFormat предлагает выбрать формат и сохраняет отчёт.
// Каждый успешно сохранённый отчёт дополнительно копируется в архив
// (logs/<timestamp>/) с манифестом и снимком журнала мониторинга.
func saveReportAnyFormat(rep report.Report, textReport string, reader *bufio.Reader) {
	utils.Println("")
	utils.Println(utils.Header("--- Выбор формата отчёта ---"))
	utils.Println(utils.Info("[1]") + " TXT (текстовый файл)")
	utils.Println(utils.Info("[2]") + " JSON")
	utils.Println(utils.Info("[3]") + " CSV (открывается в Excel)")
	utils.Println(utils.Info("[4]") + " HTML (открывается в браузере)")
	utils.PrintMenu("Выберите формат (1-4): ")

	choice, _ := reader.ReadString('\n')
	choice = strings.TrimSpace(choice)

	switch choice {
	case "2", "3", "4":
		var rep2 report.Reporter
		switch choice {
		case "2":
			rep2 = report.JSONReporter{}
		case "3":
			rep2 = report.CSVReporter{}
		case "4":
			rep2 = report.HTMLReporter{}
		}

		data, err := rep2.Write(rep)
		if err != nil {
			utils.Println(utils.ErrorText(fmt.Sprintf("Ошибка формирования отчёта: %v", err)))
		} else {
			filename := "diagnostic_report." + rep2.Ext()
			if err := os.WriteFile(filename, data, 0644); err != nil {
				utils.Println(utils.ErrorText(fmt.Sprintf("Ошибка сохранения: %v", err)))
			} else {
				utils.Println(utils.Success(fmt.Sprintf("Отчёт сохранён в: %s", filename)))
				if dir, aerr := archive.Save(data, rep2.Ext()); aerr == nil {
					utils.Println(utils.Info("[i] Копия в архиве: " + dir))
				} else {
					utils.Println(utils.Warning("[!] Архив не создан: " + aerr.Error()))
				}
			}
		}

	default:
		// "1" и любой другой ввод — сохраняем как раньше, в TXT
		filename := "diagnostic_report.txt"
		if err := saveReportToFile(filename, textReport); err != nil {
			utils.Println(utils.ErrorText(fmt.Sprintf("Ошибка сохранения: %v", err)))
		} else {
			utils.Println(utils.Success(fmt.Sprintf("Полный отчёт сохранён в: %s", filename)))
			if dir, aerr := archive.Save([]byte(textReport), "txt"); aerr == nil {
			baseline.Run(dir) // сохранение baseline-снимка

				utils.Println(utils.Info("[i] Копия в архиве: " + dir))
			} else {
				utils.Println(utils.Warning("[!] Архив не создан: " + aerr.Error()))
			}
		}
	}
}

// colorizeReport добавляет ANSI-цвета в текст отчёта при выводе в консоль.
func colorizeReport(text string) string {
	lines := strings.Split(text, "\n")
	var result strings.Builder

	for _, line := range lines {
		trimmed := strings.TrimSpace(line)

		switch {
		case strings.Contains(trimmed, "ОТЧЁТ ДИАГНОСТИКИ ПК"):
			result.WriteString(utils.Title(line) + "\n")
		case strings.HasPrefix(trimmed, "===") && strings.HasSuffix(trimmed, "==="):
			result.WriteString(utils.Header(line) + "\n")
		case trimmed == "---":
			result.WriteString(utils.Dim(line) + "\n")
		case strings.HasSuffix(trimmed, ":") && !strings.HasPrefix(trimmed, " ") && !strings.HasPrefix(trimmed, "\t"):
			result.WriteString(utils.Bold(utils.Info(line)) + "\n")
		case strings.HasSuffix(trimmed, ":") && (strings.HasPrefix(trimmed, " ") || strings.HasPrefix(trimmed, "\t")):
			result.WriteString(utils.Cyan(line) + "\n")
		case strings.Contains(trimmed, "ОШИБКА") || strings.Contains(trimmed, "Ошибка соединения"):
			result.WriteString(utils.ErrorText(line) + "\n")
		case strings.Contains(trimmed, "ПРЕДУПРЕЖДЕНИЕ"):
			result.WriteString(utils.Warning(line) + "\n")
		case strings.Contains(trimmed, "Тест скорости диска"):
			result.WriteString(utils.Bold(line) + "\n")
		default:
			result.WriteString(line + "\n")
		}
	}

	return result.String()
}

func printCompactSummary(results []hardware.DiagnosticResult) {
	utils.Println("")
	utils.Println(utils.Header("Сбор данных завершён:"))
	for _, res := range results {
		cleanVal := strings.TrimSpace(res.Value)
		lines := strings.Split(cleanVal, "\n")
		for _, line := range lines {
			line = strings.TrimSpace(line)
			if line != "" && !strings.HasSuffix(line, ":") {
				utils.Printf("- %s: %s", utils.Bold(res.Name), line)
				break
			}
		}
	}
}

// runCLIMode выполняет действия, заданные флагами, без интерактивного меню.
// Примеры: pc-diagnostic --full-test --json-report
//          pc-diagnostic --cpu --dur 120
func runCLIMode() {
	// Веб-режим: сервер сам открывает браузер, остальное не требуется.
	if *flagWeb {
		web.StartServer()
		return
	}

	dur := *flagDur
	if dur < 10 {
		dur = 10
	}
	if dur > 600 {
		dur = 600
	}

	// Стресс-тесты выполняются последовательно.
	if *flagCPU {
		utils.Println(utils.Info(fmt.Sprintf("[•] Стресс-тест CPU, %d сек...", dur)))
		utils.Println(stress.RunCPU(dur, 90))
	}
	if *flagMemory {
		utils.Println(utils.Info(fmt.Sprintf("[•] Стресс-тест RAM, %d сек...", dur)))
		utils.Println(stress.RunRAM(dur))
	}
	if *flagDisk {
		utils.Println(utils.Info(fmt.Sprintf("[•] Стресс-тест диска, %d сек...", dur)))
		utils.Println(stress.RunDisk(dur))
	}

	// Полная диагностика: собираем отчёт и сохраняем в файл.
	// --json-report подразумевает --full-test.
	if *flagFull || *flagJSON {
		utils.Println(utils.Info("[•] Полная диагностика..."))
		hwResults := hardware.RunDiagnostics()
		fullReport, structured := getFullReport(hwResults)

		var data []byte
		var err error
		ext := "txt"
		if *flagJSON {
			rep := report.JSONReporter{}
			data, err = rep.Write(structured)
			ext = rep.Ext()
		} else {
			data = []byte(fullReport)
		}

		if err != nil {
			utils.Println(utils.ErrorText(fmt.Sprintf("Ошибка формирования отчёта: %v", err)))
			os.Exit(1)
		}

		filename := "diagnostic_report." + ext
		if err := os.WriteFile(filename, data, 0644); err != nil {
			utils.Println(utils.ErrorText(fmt.Sprintf("Ошибка сохранения: %v", err)))
			os.Exit(1)
		}
		utils.Println(utils.Success("Отчёт сохранён в: " + filename))

		if dir, aerr := archive.Save(data, ext); aerr == nil {
			utils.Println(utils.Info("[i] Копия в архиве: " + dir))
			baseline.Run(dir) // сохранение baseline-снимка

		} else {
			utils.Println(utils.Warning("[!] Архив не создан: " + aerr.Error()))
		}
	}

	// Флаги заданы, но ни одно действие не выбрано — показываем справку.
	if !*flagCPU && !*flagMemory && !*flagDisk && !*flagFull && !*flagJSON && !*flagWeb {
		utils.Println(utils.Warning("Указаны параметры без действия. Доступные флаги:"))
		flag.Usage()
	}
}

// runArchivesMenu — пункт меню 9: список архивов и сравнение двух из них.
func runArchivesMenu(reader *bufio.Reader) {
	utils.Println("")
	dirs := archive.List()
	if len(dirs) == 0 {
		//utils.Println(utils.Info("[i] Архивов пока нет. Сохраните отчёт (пункт 4) — появится первый архив в папке logs/."))
		utils.Println(utils.Info("[i] Архивов пока нет. Сохраните отчёт (пункт 1, затем выбор формата 3) — появится первый архив в папке logs/."))

	} else {
		utils.Printf("Доступные архивы (%d):\n", len(dirs))
		for i, d := range dirs {
			utils.Printf("  [%d] %s\n", i+1, d)
		}
		utils.Println("")
		utils.Print("Номер первого архива для сравнения: ")
		line, _ := reader.ReadString('\n')
		a, err1 := strconv.Atoi(strings.TrimSpace(line))
		if err1 == nil && a >= 1 && a <= len(dirs) {
			utils.Print("Номер второго архива: ")
			line, _ = reader.ReadString('\n')
			b, err2 := strconv.Atoi(strings.TrimSpace(line))
			if err2 == nil && b >= 1 && b <= len(dirs) && b != a {
				archive.Compare(
					filepath.Join("logs", dirs[a-1]),
					filepath.Join("logs", dirs[b-1]),
				)
				fmt.Print(baseline.CompareDirs(
    filepath.Join("logs", dirs[a-1]),
    filepath.Join("logs", dirs[b-1]),
))

			} else {
				utils.Println(utils.Warning("Неверный номер или совпадает с первым."))
			}
		} else {
			utils.Println(utils.Warning("Неверный номер архива."))
		}
	}
	utils.Println("")
	utils.Println(utils.Dim("Нажмите Enter для возврата в меню..."))
	_, _ = reader.ReadString('\n')
}

func runConsoleMode(reader *bufio.Reader) {
	for {
		printMainMenu()
		choice, _ := reader.ReadString('\n')
		choice = strings.TrimSpace(strings.ToLower(choice))

		switch choice {
		case "1":
			start := time.Now()
			utils.Println(utils.Info("[•] Сбор данных об оборудовании..."))
			hwResults := hardware.RunDiagnostics()
			elapsed := time.Since(start)
			utils.Println(utils.Success("[✓] Оборудование проверено."))

			// Отчёт собирается один раз: текстовая и структурированная версии
			fullReport, structuredReport := getFullReport(hwResults)

			// Внутренний цикл меню вывода — работает, пока не выберут выход.
		OUTPUT_MENU:
			for {
				printOutputMenu()
				outChoice, _ := reader.ReadString('\n')
				outChoice = strings.TrimSpace(strings.ToLower(outChoice))

				switch outChoice {
				case "1":
					printCompactSummary(hwResults)
					utils.Printf("\nВремя выполнения: %v\n", elapsed.Round(time.Second))
					_, _ = reader.ReadString('\n')
					// остаёмся в этом меню

				case "2":
					utils.Println("")
					utils.Println(utils.Header("====================== НАЧАЛО ОТЧЁТА ======================"))
					utils.Println(colorizeReport(fullReport))
					utils.Println(utils.Header("======================== КОНЕЦ ОТЧЁТА ====================="))
					_, _ = reader.ReadString('\n')
					// остаёмся в этом меню

				case "3":
					saveReportAnyFormat(structuredReport, fullReport, reader)
					// после сохранения снова показываем меню —
					// можно сохранить ещё в один формат

				case "q", "0":
					break OUTPUT_MENU // вернуться в главное меню консоли

// stripFirstLine удаляет первую строку, если она совпадает с названием секции
func stripFirstLine(text string) string {
	lines := strings.SplitN(text, "\n", 2)
	if len(lines) > 1 {
		first := strings.TrimSpace(lines[0])
		if strings.HasSuffix(first, ":") ||
			(strings.HasPrefix(first, "===") && strings.HasSuffix(first, "===")) {
			return strings.TrimSpace(lines[1])
		}
	}
	return strings.TrimSpace(text)
}

	ParallelExecution: true,
	OutputFormat:    "text",
	OutputFile:      "diagnostic_report.txt",
}

// LoadFromFile загружает конфигурацию из JSON‑файла
func LoadFromFile(filename string) (AppConfig, error) {
	// Проверяем существование файла
	if _, err := os.Stat(filename); os.IsNotExist(err) {
		return Default, fmt.Errorf("файл конфигурации не найден: %s. Используется конфигурация по умолчанию", filename)
	}

	data, err := os.ReadFile(filename)
	if err != nil {
		return Default, fmt.Errorf("не удалось прочитать файл конфигурации %s: %w", filename, err)
	}

	var cfg AppConfig
	err = json.Unmarshal(data, &cfg)
	if err != nil {
		return Default, fmt.Errorf("ошибка парсинга JSON в файле %s: %w", filename, err)
	}

	// Применяем значения по умолчанию для отсутствующих полей
	if cfg.ParallelExecution == false && !isFieldPresent(data, "parallel_execution") {
		cfg.ParallelExecution = Default.ParallelExecution
	}
	if cfg.OutputFormat == "" && !isFieldPresent(data, "output_format") {
		cfg.OutputFormat = Default.OutputFormat
	}
	if cfg.OutputFile == "" && !isFieldPresent(data, "output_file") {
		cfg.OutputFile = Default.OutputFile
	}

	return cfg, nil
}

// isFieldPresent проверяет, присутствует ли поле в JSON‑данных
func isFieldPresent(jsonData []byte, fieldName string) bool {
	var data map[string]interface{}
	if err := json.Unmarshal(jsonData, &data); err != nil {
		return false
	}
	_, exists := data[fieldName]
	return exists
}

// Package thresholds реализует систему статусов Normal/Warning/Critical/Unknown
// с пороговыми значениями из config.json.
package thresholds
import (
	"encoding/json"
	"os"
	"sync"
)

// Status — статус состояния компонента.
type Status string

const (
	StatusNormal   Status = "Normal"
	StatusWarning  Status = "Warning"
	StatusCritical Status = "Critical"
	StatusUnknown  Status = "Unknown"
)

type limit struct {
	Warning  float64 `json:"warning"`
	Critical float64 `json:"critical"`
}

type thresholdConfig struct {
	CPU  limit `json:"cpu"`
	RAM  limit `json:"ram"`
	Temp limit `json:"temp"`
}

var (
	once   sync.Once
	loaded thresholdConfig
)

// load читает config.json один раз за время работы программы.
// Отсутствующие или некорректные значения заменяются значениями по умолчанию —
// программа не должна падать из-за опечатки в конфиге.
func load() thresholdConfig {
	once.Do(func() {
		loaded = thresholdConfig{
			CPU:  limit{Warning: 70, Critical: 90},
			RAM:  limit{Warning: 80, Critical: 95},
			Temp: limit{Warning: 70, Critical: 85},
		}

		data, err := os.ReadFile("config.json")
		if err != nil {
			return // файла нет — работаем на дефолтах
		}

		var wrapper struct {
			Thresholds *thresholdConfig `json:"thresholds"`
		}
		if json.Unmarshal(data, &wrapper) != nil || wrapper.Thresholds == nil {
			return // секция отсутствует или битая — дефолты
		}

		t := wrapper.Thresholds
		if t.CPU.Warning > 0 {
			loaded.CPU.Warning = t.CPU.Warning
		}
		if t.CPU.Critical > 0 {
			loaded.CPU.Critical = t.CPU.Critical
		}
		if t.RAM.Warning > 0 {
			loaded.RAM.Warning = t.RAM.Warning
		}
		if t.RAM.Critical > 0 {
			loaded.RAM.Critical = t.RAM.Critical
		}
		if t.Temp.Warning > 0 {
			loaded.Temp.Warning = t.Temp.Warning
		}
		if t.Temp.Critical > 0 {
			loaded.Temp.Critical = t.Temp.Critical
		}
	})
	return loaded
}

// eval возвращает статус по значению и порогам.
func eval(v float64, l limit) Status {
	if v >= l.Critical {
		return StatusCritical
	}
	if v >= l.Warning {
		return StatusWarning
	}
	return StatusNormal
}

// ForCPU оценивает загрузку процессора (0–100%).
func ForCPU(pct float64) Status { return eval(pct, load().CPU) }

// ForRAM оценивает занятость оперативной памяти (0–100%).
func ForRAM(pct float64) Status { return eval(pct, load().RAM) }

// ForTemp оценивает температуру CPU. Значение 0 и ниже трактуется
// как «датчик недоступен» — статус Unknown.
func ForTemp(temp float64) Status {
	if temp <= 0 {
		return StatusUnknown
	}
	return eval(temp, load().Temp)
}

// Worst сводит частные статусы в общий: критичное важнее предупреждения,
// предупреждение важнее нормы. Unknown не понижает общий статус, если
// хотя бы по одному компоненту есть данные.
func Worst(list ...Status) Status {
	hasWarning, hasUnknown, hasNormal := false, false, false
	for _, s := range list {
		switch s {
		case StatusCritical:
			return StatusCritical
		case StatusWarning:
			hasWarning = true
		case StatusUnknown:
			hasUnknown = true
		case StatusNormal:
			hasNormal = true
		}
	}
	switch {
	case hasWarning:
		return StatusWarning
	case hasNormal:
		return StatusNormal
	case hasUnknown:
		return StatusUnknown
	default:
		return StatusUnknown
	}
}

// Package scheduler — автоматический сбор метрик по расписанию
// с записью в CSV-журнал.
package scheduler

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"os/signal"
	"strconv"
	"strings"
	"time"

	"pc-diagnostic/monitor"
	"pc-diagnostic/thresholds"
)

const logFile = "schedule_log.csv"

type scheduleConfig struct {
	Enabled         bool `json:"enabled"`
	IntervalSeconds int  `json:"interval_seconds"`
}

type rootConfig struct {
	Schedule scheduleConfig `json:"schedule"`
}

// RunIfRequested проверяет, запрошен ли фоновый сбор: флагом --schedule
// или секцией "schedule": {"enabled": true} в config.json.
// Если запуск не нужен — сразу возвращает false (обычный запуск программы).
func RunIfRequested() bool {
	// --schedule имеет приоритет: если с ним переданы другие флаги — предупредим.
	

	interval, start := resolveInterval()
	if !start {
		return false
	}
	Run(interval)
	return true
}


// resolveInterval определяет интервал и факт запуска.
// Приоритет: флаг --schedule (со значением или без), затем config.json.
func resolveInterval() (time.Duration, bool) {
	flagSeen := false
	flagVal := 0
	for i := 1; i < len(os.Args); i++ {
		arg := os.Args[i]
		if arg == "--schedule" {
			flagSeen = true
			// значение может стоять отдельным аргументом: --schedule 30
			if i+1 < len(os.Args) {
				if d, err := strconv.Atoi(os.Args[i+1]); err == nil && d >= 5 {
					flagVal = d
				}
			}
			break
		}
		if strings.HasPrefix(arg, "--schedule=") {
			flagSeen = true
			if d, err := strconv.Atoi(strings.TrimPrefix(arg, "--schedule=")); err == nil && d >= 5 {
				flagVal = d
			}
			break
		}
	}

	if flagSeen {
	// Предупреждаем, только если --schedule действительно передан
		// вместе с другими флагами.
		for _, a := range os.Args[1:] {
			if strings.HasPrefix(a, "--") && !strings.HasPrefix(a, "--schedule") {
				fmt.Println("[i] Флаг --schedule имеет приоритет: остальные флаги проигнорированы.")
				break
			}
		}
		iv := 60
		if flagVal >= 5 {
			iv = flagVal
		} else if cfg, ok := readConfig(); ok && cfg.IntervalSeconds >= 5 {
			iv = cfg.IntervalSeconds
		}
		return time.Duration(iv) * time.Second, true
	}

	// Без флага — только config.json.
	if cfg, ok := readConfig(); ok && cfg.Enabled {
		iv := cfg.IntervalSeconds
		if iv < 5 {
			iv = 60
		}
		return time.Duration(iv) * time.Second, true
	}
	return 60 * time.Second, false
}

// readConfig читает секцию schedule из config.json (мягко: файл может
// отсутствовать или не содержать секции — это не ошибка).
func readConfig() (scheduleConfig, bool) {
	raw, err := os.ReadFile("config.json")
	if err != nil {
		return scheduleConfig{}, false
	}
	var cfg rootConfig
	if err := json.Unmarshal(raw, &cfg); err != nil {
		return scheduleConfig{}, false
	}
	return cfg.Schedule, true
}

// statsMap снимает метрики через monitor.GetStats и добавляет статусы —
// тот же приём, что в веб-обработчике /api/stats.
func statsMap() map[string]interface{} {
	data, _ := json.Marshal(monitor.GetStats())
	var m map[string]interface{}
	_ = json.Unmarshal(data, &m)

	getF := func(k string) float64 {
		v, _ := m[k].(float64)
		return v
	}
	stCPU := thresholds.ForCPU(getF("cpu_total"))
	stRAM := thresholds.ForRAM(getF("ram_used_pct"))
	stTemp := thresholds.ForTemp(getF("cpu_temp"))
	m["status_cpu"] = string(stCPU)
	m["status_ram"] = string(stRAM)
	m["status_temp"] = string(stTemp)
	m["status_overall"] = string(thresholds.Worst(stCPU, stRAM, stTemp))
	return m
}

func getS(m map[string]interface{}, k string) string {
	v, _ := m[k].(string)
	return v
}

func getF(m map[string]interface{}, k string) float64 {
	v, _ := m[k].(float64)
	return v
}

func csvHeader() string {
	return "time,cpu_total_pct,ram_used_pct,ram_used_gb,ram_total_gb,swap_used_pct,cpu_temp,status_cpu,status_ram,status_temp,status_overall\n"
}

func csvRow(m map[string]interface{}, now time.Time) string {
	return fmt.Sprintf("%s,%.1f,%.1f,%.2f,%.2f,%.1f,%.1f,%s,%s,%s,%s\n",
		now.Format("2006-01-02T15:04:05"),
		getF(m, "cpu_total"),
		getF(m, "ram_used_pct"),
		getF(m, "ram_used_gb"),
		getF(m, "ram_total_gb"),
		getF(m, "swap_used_pct"),
		getF(m, "cpu_temp"),
		getS(m, "status_cpu"),
		getS(m, "status_ram"),
		getS(m, "status_temp"),
		getS(m, "status_overall"),
	)
}

// colorize окрашивает статус для консоли: Normal зелёный,
// Warning жёлтый, Critical красный, Unknown серый.
func colorize(s string) string {
	colors := map[string]string{
		"Normal":   "\033[32m",
		"Warning":  "\033[33m",
		"Critical": "\033[31m",
		"Unknown":  "\033[90m",
	}
	return colors[s] + s + "\033[0m"
}

// Run запускает бесконечный сбор: первый снимок сразу, дальше по тикеру.
// Остановка — Ctrl+C, журнал корректно закрывается.
func Run(interval time.Duration) {
	fmt.Printf("[•] Сбор по расписанию: снимок каждые %v, журнал: %s\n", interval, logFile)
	fmt.Println("    Остановка: Ctrl+C")

	if _, err := os.Stat(logFile); os.IsNotExist(err) {
		if err := os.WriteFile(logFile, []byte(csvHeader()), 0o644); err != nil {
			fmt.Printf("[!] Не удалось создать журнал: %v\n", err)
			return
		}
	}

	f, err := os.OpenFile(logFile, os.O_APPEND|os.O_WRONLY, 0o644)
	if err != nil {
		fmt.Printf("[!] Не удалось открыть журнал: %v\n", err)
		return
	}
	defer f.Close()

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
	defer stop()

	snapshot := func() {
		now := time.Now()
		m := statsMap()
		if _, err := f.WriteString(csvRow(m, now)); err != nil {
			fmt.Printf("[!] Ошибка записи в журнал: %v\n", err)
			return
		}
		_ = f.Sync()
		fmt.Printf("[%s] CPU %.0f%%  RAM %.0f%%  темп %.1f°C  Статус: %s\n",
			now.Format("15:04:05"),
			getF(m, "cpu_total"),
			getF(m, "ram_used_pct"),
			getF(m, "cpu_temp"),
			colorize(getS(m, "status_overall")),
		)
	}
statsMap()                        // первый прогрев: сенсор отдаёт ноль
time.Sleep(1500 * time.Millisecond)
statsMap()                        // второй прогрев — уже с реальной температурой


	snapshot() // первый снимок сразу, без ожидания тика
	t := time.NewTicker(interval)
	defer t.Stop()
	for {
		select {
		case <-ctx.Done():
			fmt.Println("\n[✓] Сбор по расписанию остановлен. Журнал: " + logFile)
			return
		case <-t.C:
			snapshot()
		}
	}
}

// Package history — просмотр журнала мониторинга schedule_log.csv
// в виде сводки за весь период наблюдения.
package history

import (
	"encoding/csv"
	"fmt"
	"io"
	"os"
	"strconv"
	"strings"
	"time"
)

type row struct {
	t      time.Time
	cpu    float64
	ram    float64
	temp   float64
	swap   float64
	status string // status_overall
}

const logFile = "schedule_log.csv"

// SummaryText возвращает сводку по журналу строкой. Журнал может
// отсутствовать или быть пустым — это нормальный случай, не ошибка.
// Используется и консолью, и веб-интерфейсом.
func SummaryText() string {
	if _, err := os.Stat(logFile); os.IsNotExist(err) {
		return "[i] Журнал ещё не создан. Запустите сбор: pc-diagnostic.exe --schedule=60\n"
	}

	f, err := os.Open(logFile)
	if err != nil {
		return fmt.Sprintf("[!] Не удалось открыть журнал: %v\n", err)
	}
	defer f.Close()

	r := csv.NewReader(f)

	var rows []row
	for i := 0; ; i++ {
		rec, err := r.Read()
		if err == io.EOF {
			break
		}
		if err != nil {
			return fmt.Sprintf("[!] Ошибка чтения журнала: %v\n", err)
		}
		if i == 0 {
			continue // заголовок
		}
		// Формат: time,cpu_total_pct,ram_used_pct,ram_used_gb,ram_total_gb,
		//         swap_used_pct,cpu_temp,status_cpu,status_ram,status_temp,status_overall
		if len(rec) < 11 {
			continue // неполная строка — пропускаем
		}
		rr := row{status: strings.TrimSpace(rec[10])}
		if rr.t, err = time.Parse("2006-01-02T15:04:05", strings.TrimSpace(rec[0])); err != nil {
			continue
		}
		rr.cpu = atof(rec[1])
		rr.ram = atof(rec[2])
		rr.swap = atof(rec[5])
		rr.temp = atof(rec[6])
		rows = append(rows, rr)
	}

	if len(rows) == 0 {
		return "[i] Журнал пуст (нет ни одной записи).\n"
	}

	// Накопители.
	var cpuSum, ramSum, tempSum float64
	var cpuMax, ramMax, tempMax float64
	warnings, criticals := 0, 0
	for _, rr := range rows {
		cpuSum += rr.cpu
		ramSum += rr.ram
		tempSum += rr.temp
		if rr.cpu > cpuMax {
			cpuMax = rr.cpu
		}
		if rr.ram > ramMax {
			ramMax = rr.ram
		}
		if rr.temp > tempMax {
			tempMax = rr.temp
		}
		switch rr.status {
		case "Warning":
			warnings++
		case "Critical":
			criticals++
		}
	}
	n := float64(len(rows))
	first, last := rows[0].t, rows[len(rows)-1].t
	span := last.Sub(first)
	if span < 0 {
		span = 0
	}

	var b strings.Builder
	fmt.Fprintln(&b)
	fmt.Fprintln(&b, "========================================")
	fmt.Fprintln(&b, "    ИСТОРИЯ МОНИТОРИНГА (schedule_log.csv)")
	fmt.Fprintln(&b, "========================================")
	fmt.Fprintf(&b, "Период наблюдения:  %s — %s\n", first.Format("02.01.2006 15:04:05"), last.Format("15:04:05"))
	fmt.Fprintf(&b, "Длительность:       %v\n", span.Round(time.Second))
	fmt.Fprintf(&b, "Снимков:            %d\n", len(rows))
	fmt.Fprintln(&b)
	fmt.Fprintf(&b, "CPU:   среднее %.1f%%, максимум %.1f%%\n", cpuSum/n, cpuMax)
	fmt.Fprintf(&b, "RAM:   среднее %.1f%%, максимум %.1f%%\n", ramSum/n, ramMax)
	fmt.Fprintf(&b, "Температура CPU: среднее %.1f°C, максимум %.1f°C\n", tempSum/n, tempMax)
	fmt.Fprintln(&b)
	fmt.Fprintf(&b, "Статусы: Critical %d, Warning %d, Normal %d\n",
		criticals, warnings, len(rows)-warnings-criticals)

	// Вывод-оценка по счётчикам.
	switch {
	case criticals > 0:

// Save сохраняет копию отчёта в папку logs/<timestamp>/.
// reportData — содержимое отчёта, ext — расширение (txt, json, csv, html).
// Возвращает путь к созданной папке.
func Save(reportData []byte, ext string) (string, error) {
	name := time.Now().Format("2006-01-02_15-04-05")
	dir := filepath.Join("logs", name)
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return "", fmt.Errorf("не удалось создать папку архива: %w", err)
	}

	files := []string{}

	rp := filepath.Join(dir, "diagnostic_report."+ext)
	if err := os.WriteFile(rp, reportData, 0o644); err != nil {
		return "", fmt.Errorf("не удалось записать отчёт: %w", err)
	}
	files = append(files, "diagnostic_report."+ext)

	// Снимок журнала мониторинга на момент прогона: дельта между
	// двумя архивами покажет, как менялись метрики между прогонами.
	if raw, err := os.ReadFile("schedule_log.csv"); err == nil {
		lg := filepath.Join(dir, "schedule_log.csv")
		if err := os.WriteFile(lg, raw, 0o644); err != nil {
			return "", fmt.Errorf("не удалось записать снимок журнала: %w", err)
		}
		files = append(files, "schedule_log.csv")
	}

	host, _ := os.Hostname()
	m := manifest{
		Time:     time.Now().Format("2006-01-02 15:04:05"),
		Hostname: host,
		OS:       runtime.GOOS + "/" + runtime.GOARCH,
		Files:    files,
	}
	mb, _ := json.MarshalIndent(m, "", "  ")
	_ = os.WriteFile(filepath.Join(dir, "manifest.json"), mb, 0o644)

	return dir, nil
}

func compareFileText(oldDir, newDir, name string) string {
	var b strings.Builder
	a, okA := readLines(oldDir, name)
	c, okB := readLines(newDir, name)
	if !okA && !okB {
		fmt.Fprintf(&b, "\n  %s: нет ни в одном из архивов — пропущено\n", name)
		return b.String()
	}
	if !okA || !okB {
		which := "старом"
		if !okB {
			which = "новом"
		}
		fmt.Fprintf(&b, "\n  %s: отсутствует в %s архиве — сравнение невозможно\n", name, which)
		return b.String()
	}

	fmt.Fprintf(&b, "\n----------------------------------------\n")
	fmt.Fprintf(&b, "СРАВНЕНИЕ: %s\n", name)
	fmt.Fprintf(&b, "старый: %s\nновый:  %s\n", oldDir, newDir)
	fmt.Fprintf(&b, "----------------------------------------\n")

	changes := diffLines(a, c)
	if len(changes) == 0 {
		b.WriteString("  Различий не найдено.\n")
		return b.String()
	}
	const maxOut = 80
	for i, line := range changes {
		if i == maxOut {
			fmt.Fprintf(&b, "  ... и ещё %d строк различий (вывод ограничен)\n", len(changes)-maxOut)
			return b.String()
		}
		b.WriteString(line + "\n")
	}
	return b.String()
}

//***
func readLines(dir, name string) ([]string, bool) {
	raw, err := os.ReadFile(filepath.Join(dir, name))
	if err != nil {
		return nil, false
	}
	s := strings.TrimSpace(string(raw))
	if s == "" {
		return nil, true // файл есть, но пустой
	}
	return strings.Split(s, "\n"), true
}

// diffLines — построчный diff через наибольшую общую подпоследовательность.
// Ограничение: суммарно до 6000 строк (иначе файл сравнивается целиком
// как заменённый) — защита от квадратичного роста памяти.
func diffLines(a, b []string) []string {
	if len(a)+len(b) > 6000 {
		return []string{"  [!] Файлы слишком большие для построчного сравнения"}
	}

	n, m := len(a), len(b)
	// lcs[i][j] — длина наибольшей общей подпоследовательности a[i:], b[j:].
	lcs := make([][]int, n+1)
	for i := range lcs {
		lcs[i] = make([]int, m+1)
	}
	for i := n - 1; i >= 0; i-- {
		for j := m - 1; j >= 0; j-- {
			if a[i] == b[j] {
				lcs[i][j] = lcs[i+1][j+1] + 1
			} else if lcs[i+1][j] >= lcs[i][j+1] {
				lcs[i][j] = lcs[i+1][j]
			} else {
				lcs[i][j] = lcs[i][j+1]
			}
		}
	}

	var out []string
	i, j := 0, 0
	for i < n && j < m {
		switch {
		case a[i] == b[j]:
			i++
			j++
		case lcs[i+1][j] >= lcs[i][j+1]:
			out = append(out, "- "+strings.TrimRight(a[i], "\r"))
			i++
		default:
			out = append(out, "+ "+strings.TrimRight(b[j], "\r"))
			j++
		}
	}
	for ; i < n; i++ {
		out = append(out, "- "+strings.TrimRight(a[i], "\r"))
	}
	for ; j < m; j++ {
		out = append(out, "+ "+strings.TrimRight(b[j], "\r"))
	}
	return out
}

// Package baseline — структурный снимок ключевых показателей системы
// и сравнение прогонов: свободная RAM, температура CPU, открытые порты,
// автозагрузка. Снимок хранится в baseline.json (корень проекта) и копией
// в архив отчёта, поэтому дельты доступны и между любыми двумя архивами
// (консольный пункт [9], веб-страница «Архивы»).

const lastFile = "baseline.json"

// Snapshot — снимок ключевых метрик прогона.
type Snapshot struct {
	Time       string   `json:"time"`
	RamFreeGB  float64  `json:"ram_free_gb"`
	RamUsedGB  float64  `json:"ram_used_gb"`
	RamTotalGB float64  `json:"ram_total_gb"`
	SwapUsedGB float64  `json:"swap_used_gb"`
	CpuTempC   float64  `json:"cpu_temp_c"`
	Ports      []int    `json:"open_ports"`
	Autostart  []string `json:"autostart"`
}

// Take собирает текущий снимок. RAM и температура берутся из monitor.GetStats
// (тот же источник, что в веб-мониторинге), порты сканируются самостоятельно,
// автозагрузка читается из реестра через reg query.
func Take() Snapshot {
	s := Snapshot{Time: time.Now().Format("2006-01-02 15:04:05")}

	// Поля структуры monitor.Stats читаем через map, чтобы не зависеть
	// от имён полей структуры (паттерн как в web.statsHandler).
	if data, err := json.Marshal(monitor.GetStats()); err == nil {
		var m map[string]interface{}
		if err := json.Unmarshal(data, &m); err == nil {
			num := func(k string) float64 {
				v, _ := m[k].(float64)
				return v
			}
			s.RamTotalGB = num("ram_total_gb")
			s.RamUsedGB = num("ram_used_gb")
			s.RamFreeGB = s.RamTotalGB - s.RamUsedGB
			s.SwapUsedGB = num("swap_used_gb")
			s.CpuTempC = num("cpu_temp")
		}
	}

	s.Ports = scanPorts()
	s.Autostart = readAutostart()

	sort.Slice(s.Ports, func(i, j int) bool { return s.Ports[i] < s.Ports[j] })
	sort.Strings(s.Autostart)
	return s
}

// CompareText возвращает структурные дельты двух снимков открытым текстом.
func CompareText(old, new Snapshot) string {
	var b strings.Builder

	fmt.Fprintln(&b, "----------------------------------------")
	fmt.Fprintln(&b, "BASELINE: ключевые метрики")
	fmt.Fprintf(&b, "старый: %s (RAM свободно %.2f ГБ, портов %d, автозапуск %d)\n",
		old.Time, old.RamFreeGB, len(old.Ports), len(old.Autostart))
	fmt.Fprintf(&b, "новый:  %s (RAM свободно %.2f ГБ, портов %d, автозапуск %d)\n",
		new.Time, new.RamFreeGB, len(new.Ports), len(new.Autostart))
	fmt.Fprintln(&b, "----------------------------------------")

	fmt.Fprintf(&b, "RAM (свободно):      %.2f → %.2f ГБ  (%+.2f)\n",
		old.RamFreeGB, new.RamFreeGB, new.RamFreeGB-old.RamFreeGB)
	fmt.Fprintf(&b, "Swap (используется): %.2f → %.2f ГБ  (%+.2f)\n",
		old.SwapUsedGB, new.SwapUsedGB, new.SwapUsedGB-old.SwapUsedGB)

	// Температура: сравниваем только если оба датчика вернули данные.
	switch {
	case old.CpuTempC > 0 && new.CpuTempC > 0:
		fmt.Fprintf(&b, "CPU (температура):   %.0f → %.0f °C  (%+.0f)\n",
			old.CpuTempC, new.CpuTempC, new.CpuTempC-old.CpuTempC)
	default:
		fmt.Fprintln(&b, "CPU (температура):   нет данных в одном из снимков")
	}

	fmt.Fprintln(&b)
	fmt.Fprintln(&b, "Порты (localhost):")
	oldPorts := toSet(old.Ports)
	newPorts := toSet(new.Ports)
	added, removed := 0, 0
	for _, p := range new.Ports {
		if !oldPorts[p] {
			fmt.Fprintf(&b, "  [+] %d %s — порт ОТКРЫЛСЯ\n", p, portName(p))
			added++
		}
	}
	for _, p := range old.Ports {
		if !newPorts[p] {
			fmt.Fprintf(&b, "  [-] %d %s — порт закрылся\n", p, portName(p))
			removed++
		}
	}
	if added == 0 && removed == 0 {
		fmt.Fprintf(&b, "  совпадает (%d портов)\n", len(new.Ports))
	}

	fmt.Fprintln(&b)
	fmt.Fprintln(&b, "Автозагрузка:")
	oldAuto := toStrSet(old.Autostart)
	newAuto := toStrSet(new.Autostart)
	aAdded, aRemoved := 0, 0
	for _, e := range new.Autostart {
		if !oldAuto[e] {
			fmt.Fprintf(&b, "  [+] %s — новая запись\n", e)
			aAdded++
		}
	}
	for _, e := range old.Autostart {
		if !newAuto[e] {
			fmt.Fprintf(&b, "  [-] %s — запись удалена\n", e)
			aRemoved++
		}
	}
	if aAdded == 0 && aRemoved == 0 {
		fmt.Fprintf(&b, "  изменений нет (%d записей)\n", len(new.Autostart))
	}

	// Итоговый вердикт по значимым изменениям.
	fmt.Fprintln(&b)
	switch {
	case added > 0:
		fmt.Fprintln(&b, "[!] Внимание: появились новые открытые порты — проверьте, какие программы их слушают.")
	case aAdded > 0:
		fmt.Fprintln(&b, "[!] Внимание: в автозагрузке появились новые записи.")
	case removed > 0 || aRemoved > 0:
		fmt.Fprintln(&b, "[i] Часть элементов (порты/автозагрузка) исчезла по сравнению с прошлым прогоном.")
	default:
		fmt.Fprintln(&b, "[✓] Значимых изменений в портах и автозагрузке нет.")
	}
	return b.String()
}

// printColored печатает дельту с раскраской: [!] красным, [+] зелёным,

		case strings.HasPrefix(line, "  [-]"):

	}

// portNames — короткие названия для дельты (только добавленные порты).
func portName(p int) string {
	names := map[int]string{
		21: "FTP", 22: "SSH", 23: "Telnet", 25: "SMTP", 53: "DNS",
		80: "HTTP", 110: "POP3", 135: "RPC", 139: "NetBIOS", 143: "IMAP",
		443: "HTTPS", 445: "SMB", 993: "IMAPS", 995: "POP3S", 1433: "MSSQL",
		1723: "PPTP", 3306: "MySQL", 3389: "RDP", 5432: "PostgreSQL",
		5900: "VNC", 6379: "Redis", 8080: "HTTP-alt", 8443: "HTTPS-alt",
		27017: "MongoDB",
	}
	if n, ok := names[p]; ok {
		return "(" + n + ")"
	}
	return ""
}

// scanPorts проверяет стандартные порты на localhost конкурентно.
func scanPorts() []int {
	var mu sync.Mutex
	var wg sync.WaitGroup
	var open []int

	for _, p := range portList {
		wg.Add(1)
		go func(port int) {
			defer wg.Done()
			conn, err := net.DialTimeout("tcp", "127.0.0.1:"+fmt.Sprint(port), 300*time.Millisecond)
			if err == nil {
				conn.Close()
				mu.Lock()
				open = append(open, port)
				mu.Unlock()
			}
		}(p)
	}
	wg.Wait()
	return open
}

// readAutostart читает записи автозагрузки из реестра (HKCU и HKLM, ключ Run)
// через reg query. На Linux возвращает пустой список.
func readAutostart() []string {
	if runtime.GOOS != "windows" {
		return nil
	}

	var out []string
	seen := map[string]bool{}
	for _, key := range []string{
		`HKCU\Software\Microsoft\Windows\CurrentVersion\Run`,
		`HKLM\Software\Microsoft\Windows\CurrentVersion\Run`,
	} {
		res, err := exec.Command("reg", "query", key).Output()
		if err != nil {
			continue // ключа может не быть — это нормально
		}
		for _, line := range strings.Split(string(res), "\n") {
			line = strings.TrimSpace(line)
			// Формат строки reg query: "Имя    REG_SZ    значение"
			if !strings.Contains(line, "REG_SZ") && !strings.Contains(line, "REG_EXPAND_SZ") {
				continue
			}
			parts := strings.Fields(line)
			if len(parts) < 3 {
				continue
			}
			entry := parts[0] + ": " + strings.Join(parts[2:], " ")
			if !seen[entry] {
				seen[entry] = true
				out = append(out, entry)
			}
		}
	}
	return out
}

// toSet / toStrSet — вспомогательные множества для сравнения.
func toSet(nums []int) map[int]bool {
	m := make(map[int]bool, len(nums))
	for _, n := range nums {
		m[n] = true
	}
	return m
}

func toStrSet(ss []string) map[string]bool {
	m := make(map[string]bool, len(ss))
	for _, s := range ss {
		m[s] = true
	}
	return m
}

		DiagnoseGPU,
		DiagnoseTemperature,
	}

	results := make([]DiagnosticResult, len(collectors))
	var wg sync.WaitGroup

	for i, c := range collectors {
		wg.Add(1)
		go func(idx int, fn func() DiagnosticResult) {
			defer wg.Done()
			results[idx] = fn()
		}(i, c)
	}

	wg.Wait()
	return results
}

// deviceIsSSD — определение SSD по /sys/block/<dev>/queue/rotational (0 = SSD).
func deviceIsSSD(dev string) bool {
	candidates := []string{dev}
	trimmed := dev
	for len(trimmed) > 0 {
		last := trimmed[len(trimmed)-1]
		if last >= '0' && last <= '9' {
			trimmed = trimmed[:len(trimmed)-1]
		} else {
			break
		}
	}
	if trimmed != dev && trimmed != "" {
		candidates = append(candidates, trimmed)
	}
	for _, c := range candidates {
		data, err := os.ReadFile("/sys/block/" + c + "/queue/rotational")
		if err == nil {
			return strings.TrimSpace(string(data)) == "0"
		}
	}
	return false
}

// physicalDisks — список физических дисков (без loop/ram/zram/dm).
func physicalDisks() []string {
	var out []string
	entries, _ := filepath.Glob("/sys/block/*")
	for _, e := range entries {
		name := filepath.Base(e)
		if strings.HasPrefix(name, "loop") || strings.HasPrefix(name, "ram") ||
			strings.HasPrefix(name, "dm-") || strings.HasPrefix(name, "zram") || strings.HasPrefix(name, "sr") {
			continue
		}
		out = append(out, name)
	}
	return out
}

// mountedDevices — «устройство -> точка монтирования» из /proc/mounts.
func mountedDevices() map[string]string {
	m := map[string]string{}
	data, err := os.ReadFile("/proc/mounts")
	if err != nil {
		return m
	}
	for _, line := range strings.Split(string(data), "\n") {
		f := strings.Fields(line)
		if len(f) >= 3 {
			m[f[0]] = f[1]
		}
	}
	return m
}

// CheckFileSystemInfo — проверка файловой системы в режиме только чтения
// (Linux-аналог chkdsk /scan). Смонтированные разделы пропускаются:
// fsck работающего диска небезопасен. Полная проверка требует прав root.
func CheckFileSystemInfo() string {
	var b strings.Builder
	fmt.Fprintln(&b, "Проверка файловой системы (fsck -n, только чтение):")

	if os.Geteuid() != 0 {
		fmt.Fprintln(&b, "  Для проверки разделов требуются права администратора (root).")
		fmt.Fprintln(&b, "  Запустите программу с sudo — тогда несмонтированные разделы будут проверены.")
		return b.String()
	}

	// Составляем список разделов подкачки (swap) — их проверять не нужно:
	// это не файловая система с данными, а область для выгрузки памяти.
	swapDevs := map[string]bool{}
	if data, err := os.ReadFile("/proc/swaps"); err == nil {
		for _, line := range strings.Split(string(data), "\n") {
			f := strings.Fields(strings.TrimSpace(line))
			if len(f) >= 1 && strings.HasPrefix(f[0], "/dev/") {
				swapDevs[f[0]] = true
			}
		}
	}

	mounts := mountedDevices()
	checked := 0
	for _, name := range physicalDisks() {
		dev := "/dev/" + name
		// swap и смонтированные разделы пропускаем
		if swapDevs[dev] {
			continue
		}
		if _, ok := mounts[dev]; ok {
			continue
		}
		out, err := exec.Command("fsck", "-n", dev).CombinedOutput()
		if err != nil {
			fmt.Fprintf(&b, "  %s: вывод fsck:\n%s\n", name, strings.TrimSpace(string(out)))
		} else {
			fmt.Fprintf(&b, "  %s: ошибок не найдено\n", name)
		}
		checked++
	}
	if checked == 0 {
		fmt.Fprintln(&b, "  Несмонтированных проверяемых разделов не найдено")
	}
	return b.String()
}


// GetSSDWearInfo — объём записи за жизнь (TBW) и Percentage Used через smartctl.
func GetSSDWearInfo() string {
	smartPath, err := exec.LookPath("smartctl")
	if err != nil {
		return "Износ SSD (TBW): утилита smartctl не найдена (установите пакет smartmontools 7.0+)"
	}
	var b strings.Builder
	fmt.Fprintln(&b, "Износ SSD (ресурс записи):")
	found := false
	for _, name := range physicalDisks() {
		if !deviceIsSSD(name) {
			continue
		}
		out, err := exec.Command(smartPath, "-A", "-j", "/dev/"+name).Output()
		if err != nil && len(out) == 0 {
			fmt.Fprintf(&b, "  /dev/%s: smartctl вернул ошибку: %v\n", name, err)
			continue
		}
		var doc map[string]interface{}
		if err := json.Unmarshal(out, &doc); err != nil {
			fmt.Fprintf(&b, "  /dev/%s: не удалось разобрать JSON smartctl\n", name)
			continue
		}
		found = true
		fmt.Fprintf(&b, "  /dev/%s:\n", name)
		if pu, ok := numField(doc, "percentage_used"); ok {
			status := "норма"
			if pu >= 90 {
				status = "КРИТИЧНО"
			} else if pu >= 80 {
				status = "ВНИМАНИЕ"
			}
			fmt.Fprintf(&b, "    Percentage Used: %.0f%% (%s)\n", pu, status)
		}
		if du, ok := numField(doc, "data_units_written"); ok {
			fmt.Fprintf(&b, "    Записано за жизнь: %.2f ТБ (TBW)\n", du*512000/1e12)
		}
		if attrs, ok := doc["attributes"].([]interface{}); ok {
			for _, a := range attrs {
				m, ok := a.(map[string]interface{})
				if !ok {
					continue
				}
				aname, _ := m["name"].(string)
				raw, _ := m["raw"].(map[string]interface{})
				rawStr, _ := raw["value"].(string)
				switch aname {
				case "Total_LBAs_Written":
					if n, err := strconv.ParseFloat(rawStr, 64); err == nil {
						fmt.Fprintf(&b, "    Записано за жизнь: %.2f ТБ (TBW)\n", n*512/1e12)
					}
				case "Host_Writes_32MiB":
					if n, err := strconv.ParseFloat(rawStr, 64); err == nil {
						fmt.Fprintf(&b, "    Записано за жизнь: %.2f ТБ (TBW)\n", n*33554432/1e12)
					}
				}
			}
		}
	}
	if !found {
		fmt.Fprintln(&b, "  SSD-диски не обнаружены или данные недоступны")
	}
	return b.String()
}

// numField — извлечение числового поля из JSON smartctl.
func numField(doc map[string]interface{}, key string) (float64, bool) {
	v, ok := doc[key]
	if !ok {
		return 0, false
	}
	switch x := v.(type) {
	case float64:
		return x, true
	case string:
		f, err := strconv.ParseFloat(strings.TrimSpace(x), 64)
		if err == nil {
			return f, true
		}
	}
	return 0, false
}

// AnalyzeFragmentation — анализ фрагментации через e4defrag -c (ext4),
// SSD пропускаются.
func AnalyzeFragmentation() string {
	mounts := mountedDevices()
	var b strings.Builder
	fmt.Fprintln(&b, "Анализ фрагментации (только анализ, без дефрагментации):")
	analyzed, ssdSkipped := 0, 0
	for dev, mp := range mounts {
		base := filepath.Base(dev)
		if !strings.HasPrefix(dev, "/dev/sd") && !strings.HasPrefix(dev, "/dev/nvme") {
			continue
		}
		if !strings.ContainsAny(base[len(base)-1:], "123456789") {
			continue
		}
		if deviceIsSSD(base) {
			fmt.Fprintf(&b, "  %s (%s): SSD — пропущен (фрагментация не требуется)\n", dev, mp)
			ssdSkipped++
			continue
		}
		out, err := exec.Command("e4defrag", "-c", mp).CombinedOutput()
		analyzed++
		if err != nil && strings.Contains(string(out), "Permission denied") {
			fmt.Fprintf(&b, "  %s: требуются права администратора (запустите с sudo)\n", mp)
		} else if err != nil && len(out) == 0 {
			fmt.Fprintf(&b, "  %s: ошибка анализа: %v\n", mp, err)
		} else {
			fmt.Fprintf(&b, "  %s:\n%s\n", mp, strings.TrimSpace(string(out)))
		}
	}
	if analyzed == 0 && ssdSkipped == 0 {
		fmt.Fprintln(&b, "  Подходящих разделов не найдено")
	} else if analyzed == 0 && ssdSkipped > 0 {
		fmt.Fprintln(&b, "  Все разделы — SSD: фрагментация не анализируется")
	}
	return b.String()
}

// BenchmarkDiskRandom — тест случайного доступа блоками 4 КБ (чистый Go).
func BenchmarkDiskRandom() string {
	dir := os.TempDir()
	if wd, err := os.Getwd(); err == nil {
		dir = wd
	}
	path := filepath.Join(dir, "pcdiag_random_test.bin")
	const fileSize = int64(256) << 20
	f, err := os.Create(path)
	if err != nil {
		return "Тест случайного доступа 4K: не удалось создать тестовый файл: " + err.Error()
	}
	defer func() {
		f.Close()
		os.Remove(path)
	}()
	if err := f.Truncate(fileSize); err != nil {
		return "Тест случайного доступа 4K: не удалось выделить файл: " + err.Error()
	}
	const block = 4096
	buf := make([]byte, block)
	rng := rand.New(rand.NewSource(time.Now().UnixNano()))
	maxOffset := fileSize - block

	start := time.Now()
	writes := 0
	for time.Since(start) < 3*time.Second {
		off := rng.Int63n(maxOffset)
		if _, err := f.WriteAt(buf, off); err != nil {
			break
		}
		writes++
	}
	f.Sync()

	rbuf := make([]byte, block)
	start = time.Now()
	reads := 0
	for time.Since(start) < 3*time.Second {
		off := rng.Int63n(maxOffset)
		if _, err := f.ReadAt(rbuf, off); err != nil {
			break
		}
		reads++
	}
	wMBs := float64(writes) * block / 1024 / 1024 / 3
	rMBs := float64(reads) * block / 1024 / 1024 / 3
	return fmt.Sprintf("Тест случайного доступа 4K:\n  Запись: %.1f МБ/с (IOPS ~%.0f)\n  Чтение: %.1f МБ/с (IOPS ~%.0f)\n  Примечание: тест идёт через кэш ОС — сравнивайте диски между собой, а не с абсолютными эталонами", wMBs, float64(writes)/3, rMBs, float64(reads)/3)
}

// fixedDriveList возвращает список локальных фиксированных дисков ("C:", "D:", ...).
func fixedDriveList() []string {
    if runtime.GOOS != "windows" {
        return []string{"/"}
    }
    var drives []string
    for c := 'C'; c <= 'Z'; c++ {
        root := string(c) + `:\`
        if fi, err := os.Stat(root); err == nil && fi.IsDir() {
            drives = append(drives, string(c)+":")
        }
    }
    return drives
}

// withTimeout — хелпер: таймаут-контекст для консольных команд.
func withTimeout(d time.Duration) (context.Context, context.CancelFunc) {
    return context.WithTimeout(context.Background(), d)
}

// decodeOEM866 превращает вывод консольных утилит Windows из OEM-кодировки
// (CP866) в UTF-8. chkdsk и defrag при выводе в канал пишут именно в CP866,
// поэтому русские пути без декодирования превращаются в кракозябры.
func decodeOEM866(b []byte) string {
    dec, err := charmap.CodePage866.NewDecoder().Bytes(b)
    if err != nil {
        return string(b)
    }
    return string(dec)
}

// runCmdRaw запускает консольную утилиту с таймаутом и декодирует
// её вывод из CP866. Используется для chkdsk и defrag.
func runCmdRaw(timeout time.Duration, name string, args ...string) (string, error) {
    ctx, cancel := withTimeout(timeout)
    defer cancel()
    out, err := exec.CommandContext(ctx, name, args...).CombinedOutput()
    return decodeOEM866(out), err
}

// summarizeChkdsk фильтрует вывод chkdsk: оставляет только значимые строки.
// chkdsk пишет прогресс через каретный возврат (\r), и находки
// («Found corrupt...») прилипают к той же строке, что и прогресс.
// Нормализуем \r в \n, чтобы находки стали самостоятельными строками.
func summarizeChkdsk(out string) string {
    out = strings.ReplaceAll(out, "\r\n", "\n")
    out = strings.ReplaceAll(out, "\r", "\n")

    var keep []string
    for _, line := range strings.Split(out, "\n") {
        l := strings.TrimSpace(line)
        if l == "" {
            continue
        }
        low := strings.ToLower(l)
        // Прогресс-строки chkdsk не несут информации о состоянии ФС.
        if strings.HasPrefix(low, "progress:") ||
            strings.Contains(low, "eta:") {
            continue
        }
        if strings.Contains(low, "%") ||
            strings.Contains(low, "ошиб") ||
            strings.Contains(low, "corrupt") ||
            strings.Contains(low, "проблем") ||
            strings.Contains(low, "готов") ||
            strings.Contains(low, "error") ||
            strings.Contains(low, "found") ||
            strings.Contains(low, "both own") ||
            strings.Contains(low, "bad index") ||
            strings.Contains(low, "unneeded link") ||
            strings.Contains(low, "records processed") ||
            strings.Contains(low, "байт") {
            keep = append(keep, "  "+l)
        }
    }
    if len(keep) == 0 {
        return "  (строк с проблемами нет — серьёзных ошибок не обнаружено)"
    }
    return strings.Join(keep, "\n")
}

// CheckFileSystemInfo проверяет файловые системы всех фиксированных дисков
// в режиме «только чтение» (chkdsk без ключей восстановления — безопасно).
func CheckFileSystemInfo() string {
    if runtime.GOOS != "windows" {
        return "Проверка файловой системы: доступно только на Windows"
    }

    var b strings.Builder
    b.WriteString("Проверка файловой системы (chkdsk, режим только чтения):\n")
    b.WriteString("Безопасно: ошибки не исправляются, диск только анализируется.\n")

    for _, d := range fixedDriveList() {
        b.WriteString(fmt.Sprintf("\nДиск %s:\n", d))
        out, err := runCmdRaw(10*time.Minute, "chkdsk", d, "/scan")
        low := strings.ToLower(out)
        switch {
        case err == nil:
            b.WriteString(summarizeChkdsk(out) + "\n")
        case strings.Contains(low, "corrupt") ||
            strings.Contains(low, "ошиб") ||
            strings.Contains(low, "error"):
            // chkdsk возвращает ненулевой код выхода, когда НАШЁЛ ошибки.
            // Это не сбой команды, а результат проверки.
            b.WriteString("  ОБНАРУЖЕНЫ ПРОБЛЕМЫ ФАЙЛОВОЙ СИСТЕМЫ:\n")
            b.WriteString(summarizeChkdsk(out) + "\n")
            b.WriteString("  Рекомендация: chkdsk " + d + " /f — исправление при перезагрузке.\n")
            b.WriteString("  Автоматически не запускаем: исправление требует размонтирования тома.\n")
        default:
            b.WriteString("  Не удалось выполнить проверку: " + strings.TrimSpace(out) + "\n")
        }
    }
    return b.String()
}

type smartScanOut struct {
    Devices []struct {
        Name string `json:"name"`
        Type string `json:"type"`
    } `json:"devices"`
}

type smartAttrOut struct {
    Device struct {
        Name      string `json:"name"`
        ModelName string `json:"model_name"`
    } `json:"device"`
    NVMe *struct {
        PercentageUsed   int   `json:"percentage_used"`
        DataUnitsWritten int64 `json:"data_units_written"`
    } `json:"nvme_smart_health_information_log"`
    ATA *struct {
        Table []struct {
            ID   int    `json:"id"`
            Name string `json:"name"`
            Val  int    `json:"value"`
            Raw  struct {
                Value  int64  `json:"value"`
                String string `json:"string"`
            } `json:"raw"`
        } `json:"table"`
    } `json:"ata_smart_attributes"`
}

// GetSSDWearInfo оценивает объём записанных на SSD данных (TBW).
// Остаток ресурса здесь не дублируем — он уже есть в базовом SMART-отчёте.
func GetSSDWearInfo() string {
    if runtime.GOOS != "windows" {
        return "Оценка износа SSD: доступно только на Windows"
    }

    smartctl, _ := exec.LookPath("smartctl")
    if smartctl == "" {
        return "Оценка износа SSD: утилита 'smartctl' не найдена. Скачайте Smartmontools (smartmontools.org)."
    }

    ctx, cancel := withTimeout(30 * time.Second)
    scanBytes, err := exec.CommandContext(ctx, smartctl, "--scan", "-j").Output()
    cancel()
    if err != nil {
        return "Оценка износа SSD: не удалось получить список дисков (smartctl --scan)"
    }
    var scan smartScanOut
    if err := json.Unmarshal(scanBytes, &scan); err != nil {
        return "Оценка износа SSD: неожиданный вывод smartctl --scan (нужна версия 7.0+)"
    }

    var b strings.Builder
    b.WriteString("Износ SSD (объём записи по SMART):\n")

    for _, dev := range scan.Devices {
        ctx, cancel := withTimeout(30 * time.Second)
        out, err := exec.CommandContext(ctx, smartctl, "-A", "-j", dev.Name).Output()
        cancel()
        if err != nil {
            b.WriteString(fmt.Sprintf("\nДиск %s: данные SMART недоступны\n", dev.Name))
            continue
        }

        var s smartAttrOut
        if err := json.Unmarshal(out, &s); err != nil {
            b.WriteString(fmt.Sprintf("\nДиск %s: не удалось разобрать вывод smartctl\n", dev.Name))
            continue
        }

        b.WriteString(fmt.Sprintf("\nДиск %s (%s):\n", dev.Name, s.Device.ModelName))

        if s.NVMe != nil {
            tbw := float64(s.NVMe.DataUnitsWritten) * 512000 / 1e12
            b.WriteString(fmt.Sprintf("  Записано за жизнь (TBW): %.2f ТБ\n", tbw))
            b.WriteString(fmt.Sprintf("  Использовано ресурса:    %d%%\n", s.NVMe.PercentageUsed))
            if remain := 100 - s.NVMe.PercentageUsed; remain < 10 {
                b.WriteString("  ВНИМАНИЕ: ресурс почти исчерпан, планируйте замену\n")
            }
        } else if s.ATA != nil {
            var tbw float64
            for _, a := range s.ATA.Table {
                if a.ID != 241 { // Total_LBAs_Written / Host_Writes_32MiB
                    continue
                }
                n, e := strconv.ParseInt(strings.TrimSpace(a.Raw.String), 10, 64)
                if e != nil || n <= 0 {
                    n = a.Raw.Value
                }
                if n <= 0 {
                    continue
                }
                // Единицы зависят от прошивки. Если имя атрибута сообщает
                // о блоках 32 МиБ (Kingston и др.) — считаем в них,
                // иначе это классические LBA по 512 байт.
                if strings.Contains(a.Name, "32MiB") {
                    tbw = float64(n) * 32 * 1024 * 1024 / 1e12
                } else {
                    tbw = float64(n) * 512 / 1e12
                }
            }
            if tbw > 0 {
                b.WriteString(fmt.Sprintf("  Записано за жизнь (TBW): %.2f ТБ\n", tbw))
            } else {
                b.WriteString("  Записано за жизнь (TBW): атрибут не найден\n")
            }
        }
    }
    return b.String()
}

// driveMediaTypes возвращает соответствие «буква диска → тип носителя»
// ("SSD"/"HDD"/...) через PowerShell: Get-Partition даёт номер физического
// диска, Get-PhysicalDisk — его MediaType. nil — определить не удалось
// (PowerShell недоступен) — тогда анализ работает как раньше, по всем дискам.
func driveMediaTypes() map[string]string {
    diskOf, err := exec.Command("powershell", "-NoProfile", "-Command",
        `Get-Partition | Where-Object DriveLetter | ForEach-Object { "{0}|{1}" -f $_.DriveLetter, $_.DiskNumber }`).Output()
    if err != nil {
        return nil
    }
    mediaOf, err := exec.Command("powershell", "-NoProfile", "-Command",
        `Get-PhysicalDisk | ForEach-Object { "{0}|{1}" -f $_.DeviceId, $_.MediaType }`).Output()
    if err != nil {
        return nil
    }

    mediaByDisk := map[string]string{}
    for _, line := range strings.Split(string(mediaOf), "\n") {
        parts := strings.SplitN(strings.TrimSpace(line), "|", 2)
        if len(parts) == 2 && parts[0] != "" {
            mediaByDisk[parts[0]] = parts[1]
        }
    }

    res := map[string]string{}
    for _, line := range strings.Split(string(diskOf), "\n") {
        parts := strings.SplitN(strings.TrimSpace(line), "|", 2)
        if len(parts) == 2 && parts[0] != "" {
            res[parts[0]] = mediaByDisk[parts[1]]
        }
    }
    if len(res) == 0 {
        return nil
    }
    return res
}

// AnalyzeFragmentation запускает анализ фрагментации (defrag /A — только
// анализ, без дефрагментации). Требует прав администратора.
// SSD-диски пропускаются: дефрагментация им не нужна (контроллер сам
// распределяет записи), а запуск defrag расходует ресурс ячеек.
func AnalyzeFragmentation() string {
    if runtime.GOOS != "windows" {
        return "Анализ фрагментации: доступно только на Windows"
    }

    var b strings.Builder
    b.WriteString("Анализ фрагментации (defrag /A — только анализ):\n")

    if media == nil {
        b.WriteString("  Не удалось определить тип носителя (Get-PhysicalDisk недоступен) —\n")
        b.WriteString("  анализ выполняется по всем фиксированным дискам, как раньше.\n")
    }

    anyRun := false
    for _, d := range fixedDriveList() {
        b.WriteString(fmt.Sprintf("\nДиск %s:\n", d))

        if media != nil {
            if m, ok := media[strings.TrimSuffix(d, ":")]; ok && strings.EqualFold(m, "SSD") {
                b.WriteString("  Пропущено: диск SSD — фрагментация для него нерелевантна,\n")
                b.WriteString("  а анализ defrag расходует ресурс записи ячеек.\n")
                continue
            }
        }
        anyRun = true

        out, err := runCmdRaw(5*time.Minute, "defrag", d, "/A", "/V")
        low := strings.ToLower(out)
        switch {
        case strings.Contains(low, "отказано") || strings.Contains(low, "denied"):
            b.WriteString("  Требуются права администратора — анализ пропущен.\n")
        case strings.TrimSpace(out) == "":
            if err != nil {
                b.WriteString(fmt.Sprintf("  Утилита не вернула данных (код выхода: %v).\n", err))
            } else {
                b.WriteString("  Утилита не вернула данных. Обычно это значит, что\n")
                b.WriteString("  анализ системного диска требует прав администратора.\n")
            }
        default:
            b.WriteString(out + "\n")
        }
    }

    if media != nil && !anyRun {
        b.WriteString("\nМагнитных дисков (HDD) не найдено: в системе только SSD — анализ завершён.\n")
    }
    return b.String()
}

// measureRandom4K выполняет тест на одном диске.
func measureRandom4K(dir string) (wMbps, rMbps float64, err error) {
    f, err := os.CreateTemp(dir, "pcdiag_rand_*.tmp")
    if err != nil {
        return 0, 0, err
    }
    name := f.Name()
    defer func() {
        f.Close()
        os.Remove(name)
    }()

    const blockSize = 4096
    const fileMB = 64
    size := int64(fileMB) * 1024 * 1024
    if err := f.Truncate(size); err != nil {
        return 0, 0, err
    }

    buf := make([]byte, blockSize)
    rnd := rand.New(rand.NewSource(time.Now().UnixNano()))

    // Запись
    deadline := time.Now().Add(3 * time.Second)
    ops := 0
    for time.Now().Before(deadline) {
        off := rnd.Int63n(fileMB) * blockSize
        if _, err := f.WriteAt(buf, off); err != nil {
            return 0, 0, err
        }
        ops++
    }
    _ = f.Sync()
    wMbps = float64(ops) * blockSize / (1024 * 1024) / 3.0

    // Чтение
    deadline = time.Now().Add(3 * time.Second)
    ops = 0
    for time.Now().Before(deadline) {
        off := rnd.Int63n(fileMB) * blockSize
        if _, err := f.ReadAt(buf, off); err != nil {
            return 0, 0, err
        }
        ops++
    }
    rMbps = float64(ops) * blockSize / (1024 * 1024) / 3.0

    return wMbps, rMbps, nil
}

// diagnoseSingleDisk запускает smartctl и возвращает читаемый результат
func diagnoseSingleDisk(devicePath string) DiagnosticResult {
    cmd := exec.Command("smartctl", "-a", "-d", "auto", devicePath, "--json=c")
    var out bytes.Buffer
    cmd.Stdout = &out
    cmd.Stderr = &out

    err := cmd.Run()
    jsonOutput := out.String()
    summary := parseSMARTJSON(jsonOutput)

    if err != nil {
        if exitErr, ok := err.(*exec.ExitError); ok {
            code := exitErr.ExitCode()
            if summary != "" {
                return DiagnosticResult{Name: devicePath, Value: summary, Error: nil}
            }
            return DiagnosticResult{
                Name:  devicePath,
                Value: jsonOutput,
                Error: fmt.Errorf("smartctl вернул код %d", code),
            }
        }
        return DiagnosticResult{Name: devicePath, Value: jsonOutput, Error: err}
    }

    return DiagnosticResult{Name: devicePath, Value: summary, Error: nil}
}

// parseSMARTJSON извлекает ключевые параметры из JSON-вывода smartctl
func parseSMARTJSON(jsonStr string) string {
    getJSONValue := func(key string) string {
        search := "\"" + key + "\":"
        idx := strings.Index(jsonStr, search)
        if idx == -1 {
            return ""
        }
        start := idx + len(search)
        for start < len(jsonStr) && jsonStr[start] == ' ' {
            start++
        }
        if start >= len(jsonStr) {
            return ""
        }
        if jsonStr[start] == '"' {
            end := start + 1
            for end < len(jsonStr) && jsonStr[end] != '"' {
                end++
            }
            return jsonStr[start+1:end]
        }
        end := start
        for end < len(jsonStr) && jsonStr[end] != ',' && jsonStr[end] != '}' && jsonStr[end] != '\n' {
            end++
        }
        return strings.TrimSpace(jsonStr[start:end])
    }

    getNestedValue := func(parent, key string) string {
        idx := strings.Index(jsonStr, "\""+parent+"\":{")
        if idx == -1 {
            return ""
        }
        sub := jsonStr[idx:]
        search := "\"" + key + "\":"
        subIdx := strings.Index(sub, search)
        if subIdx == -1 {
            return ""
        }
        start := subIdx + len(search)
        for start < len(sub) && sub[start] == ' ' {
            start++
        }
        if start >= len(sub) {
            return ""
        }
        if sub[start] == '"' {
            end := start + 1
                      for end < len(sub) && sub[end] != '"' {
                end++
            }
            return sub[start+1 : end]
        }
        end := start
        for end < len(sub) && sub[end] != ',' && sub[end] != '}' && sub[end] != '\n' {
            end++
        }
        return strings.TrimSpace(sub[start:end])
    }

    var sb strings.Builder

    model := getJSONValue("model_name")
    if model != "" {
        sb.WriteString(fmt.Sprintf("  Модель: %s\n", model))
    }
    serial := getJSONValue("serial_number")
    if serial != "" {
        sb.WriteString(fmt.Sprintf("  Серийный номер: %s\n", serial))
    }
    fw := getJSONValue("firmware_version")
    if fw != "" {
        sb.WriteString(fmt.Sprintf("  Прошивка: %s\n", fw))
    }

    // Тип носителя: SSD или HDD (см. isSSDDevice ниже)
    if isSSDDevice(jsonStr) {
        sb.WriteString("  Тип: SSD\n")
    }

    capBytes := getJSONValue("bytes")
    if capBytes != "" {
        gb := parseFloat64(capBytes) / (1024 * 1024 * 1024)
        if gb > 0 {
            sb.WriteString(fmt.Sprintf("  Ёмкость: %.0f ГБ\n", gb))
        }
    }

    passed := getNestedValue("smart_status", "passed")
    if passed == "true" {
        sb.WriteString("  SMART: PASSED (диск здоров)\n")
    } else if passed == "false" {
        sb.WriteString("  SMART: FAILED (диск неисправен!)\n")
    }

    temp := getNestedValue("temperature", "current")
    if temp != "" {
        sb.WriteString(fmt.Sprintf("  Температура: %s°C\n", temp))
    }

    hours := getNestedValue("power_on_time", "hours")
    if hours != "" {
        sb.WriteString(fmt.Sprintf("  Время работы: %s часов\n", hours))
    }

    cycles := getJSONValue("power_cycle_count")
    if cycles != "" {
        sb.WriteString(fmt.Sprintf("  Циклов включения: %s\n", cycles))
    }

    endurance := getNestedValue("endurance_used", "current_percent")
    if endurance != "" {
        sb.WriteString(fmt.Sprintf("  Износ SSD: %s%%\n", endurance))
    }

    // Percentage Used (NVMe): сколько ресурса записи использовано.
    // Пороги: > 80% — внимание, > 90% — критично.
    pctUsed := getNestedValue("nvme_smart_health_information_log", "percentage_used")
    if pctUsed != "" {
        sb.WriteString(fmt.Sprintf("  Использовано ресурса (Percentage Used): %s%%\n", pctUsed))
        if p, e := strconv.Atoi(pctUsed); e == nil {
            switch {
            case p >= 90:
                sb.WriteString("  КРИТИЧНО: ресурс почти исчерпан — планируйте замену SSD\n")
            case p >= 80:
                sb.WriteString("  ВНИМАНИЕ: использовано более 80% ресурса\n")
            }
        }
    }

    remainingPerc := extractAttrValue(jsonStr, "Remaining_Lifetime_Perc")
    if remainingPerc != "" {
        sb.WriteString(fmt.Sprintf("  Остаток ресурса: %s%%\n", remainingPerc))
    }

    ssdLife := extractAttrValue(jsonStr, "SSD_Life_Left")
    if ssdLife != "" {
        sb.WriteString(fmt.Sprintf("  SSD Life Left: %s%%\n", ssdLife))
        if v, e := strconv.Atoi(strings.TrimSpace(ssdLife)); e == nil {
            switch {
            case v <= 10:
                sb.WriteString("  КРИТИЧНО: осталось менее 10% ресурса\n")
            case v <= 20:
                sb.WriteString("  ВНИМАНИЕ: осталось менее 20% ресурса\n")
            }
        }
    }

    realloc := extractAttrValue(jsonStr, "Reallocated_Sector_Ct")
    if realloc != "" {
        sb.WriteString(fmt.Sprintf("  Переназначенные сектора: %s\n", realloc))
    }

    crcErrors := extractAttrValue(jsonStr, "UDMA_CRC_Error_Count")
    if crcErrors != "" {
        sb.WriteString(fmt.Sprintf("  Ошибки передачи (UDMA CRC): %s\n", crcErrors))
    }

    result := sb.String()
    if strings.TrimSpace(result) == "" {
        return "  Не удалось извлечь данные SMART\n"
    }
    return result
}

// isSSDDevice определяет SSD по JSON-выводу smartctl:
// - наличие блока nvme_smart_health_information_log (NVMe — всегда SSD);
// - rotation_rate == 0 (SATA SSD; у магнитных дисков — 5400/7200).
// Если ничего не найдено, возвращает false — тип считается неопределённым.
func isSSDDevice(jsonStr string) bool {
    if strings.Contains(jsonStr, `"nvme_smart_health_information_log"`) {
        return true
    }
    const key = `"rotation_rate":`
    idx := strings.Index(jsonStr, key)
    if idx == -1 {
        return false
    }
    rest := strings.TrimLeft(jsonStr[idx+len(key):], " \t")
    return strings.HasPrefix(rest, "0") || strings.HasPrefix(rest, `"0`)
}

// extractAttrValue ищет raw_value для атрибута по имени
func extractAttrValue(jsonStr, attrName string) string {
    search := "\"name\":\"" + attrName + "\""
    idx := strings.Index(jsonStr, search)
    if idx == -1 {
        return ""
    }
    sub := jsonStr[idx:]
    rawSearch := "\"raw\":{\"value\":"
    rawIdx := strings.Index(sub, rawSearch)
    if rawIdx == -1 {
        return ""
    }
    start := rawIdx + len(rawSearch)
    end := start
    for end < len(sub) && sub[end] != ',' && sub[end] != '}' {
        end++
    }
    return strings.TrimSpace(sub[start:end])
}

// parseFloat64 — простой парсер float
func parseFloat64(s string) float64 {
    var f float64
    fmt.Sscanf(s, "%f", &f)
    return f
}

// BenchmarkDisk измеряет скорость линейного чтения/записи
func BenchmarkDisk(sizeMB int) (writeSpeed float64, readSpeed float64, err error) {
    writeDir := os.Getenv("TEMP")
    if writeDir == "" {
        writeDir = "."
    }
    tempFile := filepath.Join(writeDir, "pcdiag_benchmark.tmp")

    buffer := make([]byte, 32*1024)
    sizeBytes := int64(sizeMB * 1024 * 1024)

    fWrite, err := os.OpenFile(tempFile, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0644)
    if err != nil {
        return 0, 0, fmt.Errorf("ошибка создания файла для записи: %w", err)
    }

    startWrite := time.Now()
    written := int64(0)
    for written < sizeBytes {
        n, werr := fWrite.Write(buffer)
        if werr != nil {
            fWrite.Close()
            return 0, 0, fmt.Errorf("ошибка записи: %w", werr)
        }
        written += int64(n)
    }
    writeDuration := time.Since(startWrite).Seconds()
    fWrite.Close()

    if writeDuration > 0 {
        writeSpeed = float64(sizeMB) / writeDuration
    }

    fRead, err := os.Open(tempFile)
    if err != nil {
        _ = os.Remove(tempFile)
        return writeSpeed, 0, fmt.Errorf("ошибка открытия файла для чтения: %w", err)
    }

    defer func() {
        fRead.Close()
        os.Remove(tempFile)
    }()

    readBuffer := make([]byte, 32*1024)
    toRead := sizeBytes
    startRead := time.Now()
    for toRead > 0 {
        n, rerr := fRead.Read(readBuffer)
        if rerr != nil && !strings.Contains(rerr.Error(), "EOF") {
            return writeSpeed, 0, fmt.Errorf("ошибка чтения: %w", rerr)
        }
        toRead -= int64(n)
    }
    readDuration := time.Since(startRead).Seconds()
    if readDuration > 0 {
        readSpeed = float64(sizeMB) / readDuration
    }

    return writeSpeed, readSpeed, nil
}

// listBlockDevices определяет список физических дисков системы
func listBlockDevices() ([]string, error) {
    switch runtime.GOOS {
    case "windows":
        cmd := exec.Command("smartctl", "--scan")
        var out bytes.Buffer
        cmd.Stdout = &out

        if err := cmd.Run(); err != nil {
            psCmd := exec.Command("powershell", "-Command", "(Get-PhysicalDisk).DeviceId")
            var psOut bytes.Buffer
            psCmd.Stdout = &psOut
            if err := psCmd.Run(); err != nil {
                return nil, fmt.Errorf("ошибка определения дисков: %v", err)
            }
            lines := strings.Split(strings.TrimSpace(psOut.String()), "\r\n")
            var disks []string
            for _, line := range lines {
                line = strings.TrimSpace(line)
                if line != "" {
                    disks = append(disks, fmt.Sprintf("\\\\.\\PhysicalDrive%s", line))
                }
            }
            return disks, nil
        }

        lines := strings.Split(strings.TrimSpace(out.String()), "\n")
        var disks []string
        for _, line := range lines {
            line = strings.TrimSpace(line)
            if line == "" || strings.HasPrefix(line, "#") {
                continue
            }
            parts := strings.Fields(line)
            if len(parts) > 0 {
                disks = append(disks, parts[0])
            }
        }
        return disks, nil

    case "linux":
        out, err := exec.Command("lsblk", "-d", "-n", "-o", "NAME,TYPE").Output()
        if err != nil {
            return nil, err
        }
        lines := strings.Split(string(out), "\n")
        var disks []string
        for _, line := range lines {
            if strings.Contains(line, " disk") {
                parts := strings.Fields(line)
                if len(parts) > 0 {
                    disks = append(disks, "/dev/"+parts[0])
                }
            }
        }
        return disks, nil
    case "darwin":
        return []string{"/dev/disk0"}, nil
    default:
        return nil, fmt.Errorf("unsupported OS")
    }
}

// DiagnoseDisk выполняет диагностику дисков для Unix‑систем
func DiagnoseDisk() DiagnosticResult {
	result := DiagnosticResult{Name: "Диски (Unix-системы)"}
	dir := "/" // Проверяем корневой раздел вместо текущей директории

	var stat syscall.Statfs_t
	err := syscall.Statfs(dir, &stat)
	if err != nil {
		result.Error = fmt.Errorf("ошибка получения информации о диске: %v", err)
		return result
	}

	// Расчёт объёмов в байтах
	total := uint64(stat.Blocks) * uint64(stat.Bsize)
	free := uint64(stat.Bavail) * uint64(stat.Bsize)
	used := total - free

	// Форматирование в гигабайтах с округлением до 2 знаков
	gb := 1024.0 * 1024.0 * 1024.0

	var resultBuilder strings.Builder
	resultBuilder.WriteString(fmt.Sprintf("Общее пространство: %.2f ГБ\n", float64(total)/gb))
	resultBuilder.WriteString(fmt.Sprintf("Свободное место: %.2f ГБ\n", float64(free)/gb))
	resultBuilder.WriteString(fmt.Sprintf("Используемое место: %.2f ГБ", float64(used)/gb))

	// Дополнительная информация о директории
	info, err := os.Stat(dir)
	if err == nil {
		resultBuilder.WriteString(fmt.Sprintf(
			"\n\nТекущая директория: %s\nРазмер: %d байт\nВремя модификации: %s",
			dir,
			info.Size(),
			info.ModTime().Format("2006-01-02 15:04:05"),
		))
	} else {
		resultBuilder.WriteString(fmt.Sprintf("\n\nНе удалось получить информацию о директории %s: %v", dir, err))
	}

	result.Value = resultBuilder.String()
	return result
}

// DiagnoseDisk получает информацию о дисках в системе
func DiagnoseDisk() DiagnosticResult {
	result := DiagnosticResult{Name: "Диски"}
	var sb strings.Builder

	//sb.WriteString("Диски:\n")

	partitions, err := disk.Partitions(true)
	if err != nil {
		result.Error = fmt.Errorf("ошибка получения информации о разделах: %v", err)
		return result
	}

	for _, partition := range partitions {
		usage, err := disk.Usage(partition.Mountpoint)
		if err != nil {
			sb.WriteString(fmt.Sprintf("  Раздел %s: ошибка получения информации об использовании: %v\n",
				partition.Device, err))
		continue
	}

	totalGB := float64(usage.Total) / 1024 / 1024 / 1024
	freeGB := float64(usage.Free) / 1024 / 1024 / 1024
	usedGB := float64(usage.Used) / 1024 / 1024 / 1024

	// Ручной расчёт процентов — в gopsutil нет полей FreePercent/UsedPercent
	freePercent := (float64(usage.Free) / float64(usage.Total)) * 100
	usedPercent := (float64(usage.Used) / float64(usage.Total)) * 100

	sb.WriteString(fmt.Sprintf("  Устройство: %s\n", partition.Device))
	sb.WriteString(fmt.Sprintf("    Точка монтирования: %s\n", partition.Mountpoint))
	sb.WriteString(fmt.Sprintf("    Тип файловой системы: %s\n", partition.Fstype))
	sb.WriteString(fmt.Sprintf("    Всего: %.2f ГБ\n", totalGB))
	sb.WriteString(fmt.Sprintf("    Свободно: %.2f ГБ (%.2f%%)\n", freeGB, freePercent))
	sb.WriteString(fmt.Sprintf("    Использовано: %.2f ГБ (%.2f%%)\n", usedGB, usedPercent))

	// Предупреждение при низком свободном месте
	if freePercent < 10 {
		sb.WriteString("    ПРЕДУПРЕЖДЕНИЕ: мало свободного места (<10%)!\n")
	}

	sb.WriteString("\n")
	}

	result.Value = sb.String()
	return result
}

// DiagnoseMemory выполняет расширенную диагностику памяти с информацией о файле подкачки и предупреждениями
func DiagnoseMemory() DiagnosticResult {
	result := DiagnosticResult{Name: "Память"}
	var sb strings.Builder
	var wg sync.WaitGroup
	wg.Add(2)

	// Каналы для получения результатов
	vmChan := make(chan *mem.VirtualMemoryStat, 1)
	swapChan := make(chan *mem.SwapMemoryStat, 1)
	errChan := make(chan error, 2)

	// Параллельное получение данных о RAM
	go func() {
		defer wg.Done()
		vm, err := mem.VirtualMemory()
		if err != nil {
			errChan <- fmt.Errorf("ошибка получения информации о памяти: %v", err)
			return
		}
		vmChan <- vm
	}()

	// Параллельное получение данных о swap
	go func() {
		defer wg.Done()
		swap, err := mem.SwapMemory()
		if err != nil {
			errChan <- fmt.Errorf("ошибка получения информации о файле подкачки: %v", err)
			return
		}
		swapChan <- swap
	}()

	wg.Wait()
	close(vmChan)
	close(swapChan)
	close(errChan)

	// Обработка ошибок
	if len(errChan) > 0 {
		var errors []string
		for err := range errChan {
			errors = append(errors, err.Error())
		}
		result.Error = fmt.Errorf(strings.Join(errors, "; "))
		return result
	}

	// Получение результатов
	vm := <-vmChan
	var swap *mem.SwapMemoryStat
	select {
	case swap = <-swapChan:
	default:
		swap = nil
	}

	// Конвертируем байты в гигабайты
	totalRAM := float64(vm.Total) / 1024 / 1024 / 1024
	freeRAM := float64(vm.Free) / 1024 / 1024 / 1024
	availableRAM := float64(vm.Available) / 1024 / 1024 / 1024

	var totalSwapGB, usedSwapGB float64
	var swapUsagePercent float64

	if swap != nil {
		totalSwapGB = float64(swap.Total) / 1024 / 1024 / 1024
		usedSwapGB = float64(swap.Used) / 1024 / 1024 / 1024
		swapUsagePercent = swap.UsedPercent
	}

	// Выводим полную информацию о памяти
	sb.WriteString(fmt.Sprintf("  Всего оперативной памяти (RAM): %.2f ГБ\n", totalRAM))
	sb.WriteString(fmt.Sprintf("  Свободно (Free): %.2f ГБ\n", freeRAM))
	sb.WriteString(fmt.Sprintf("  Доступно (Available, включая кэши): %.2f ГБ\n", availableRAM))
	sb.WriteString(fmt.Sprintf("  Используется RAM: %.2f%%\n", vm.UsedPercent))

	// Информация о файле подкачки
	if swap != nil {
		sb.WriteString("\n  Файл подкачки (pagefile):\n")
		sb.WriteString(fmt.Sprintf("    Всего: %.2f ГБ\n", totalSwapGB))
		sb.WriteString(fmt.Sprintf("    Использовано: %.2f ГБ\n", usedSwapGB))
		sb.WriteString(fmt.Sprintf("    Использование: %.2f%%\n", swapUsagePercent))
	} else {
		sb.WriteString("\n  Файл подкачки: не обнаружен или недоступен\n")
	}

	// Предупреждения при критическом использовании
	if vm.UsedPercent > 90 {
		sb.WriteString("\n  ПРЕДУПРЕЖДЕНИЕ: высокая загрузка оперативной памяти (>90%!)\n")
	}
	if swap != nil && swapUsagePercent > 80 {
		sb.WriteString("  ПРЕДУПРЕЖДЕНИЕ: высокое использование файла подкачки (>80%!)\n")
	}

	result.Value = sb.String()
	return result
}

// getDNSServers получает список DNS‑серверов
func getDNSServers() ([]string, error) {
	var servers []string

	if runtime.GOOS == "windows" {
		cmd := exec.Command("powershell", "-Command",
			"(Get-DnsClientServerAddress -AddressFamily IPv4).ServerAddresses")
		output, err := cmd.Output()
		if err != nil {
			return nil, fmt.Errorf("ошибка выполнения PowerShell команды: %v", err)
	}

		lines := strings.Split(strings.TrimSpace(string(output)), "\n")
		for _, line := range lines {
			line = strings.TrimSpace(line)
			if line != "" {
				servers = append(servers, line)
		}
	}
	} else {
		// Linux: читаем /etc/resolv.conf
		data, err := os.ReadFile("/etc/resolv.conf")
		if err != nil {
			return nil, err
	}

		lines := strings.Split(string(data), "\n")
		for _, line := range lines {
			if strings.HasPrefix(line, "nameserver") {
				parts := strings.Fields(line)
				if len(parts) > 1 {
					servers = append(servers, parts[1])
			}
		}
	}
	}

	if len(servers) == 0 {
		return nil, fmt.Errorf("DNS‑серверы не найдены")
	}
	return servers, nil
}

func GetSensorsInfo() string {
	dirs, _ := filepath.Glob("/sys/class/hwmon/hwmon*")
	var b strings.Builder
	fans, volts := 0, 0
	for _, dir := range dirs {
		chip := hwmonFile(dir, "name")
		fanFiles, _ := filepath.Glob(dir + "/fan*_input")
		for _, f := range fanFiles {
			if v := hwmonFile(dir, filepath.Base(f)); v != "" {
				if rpm, err := strconv.Atoi(v); err == nil && rpm > 0 {
					label := hwmonFile(dir, strings.TrimSuffix(filepath.Base(f), "_input")+"_label")
					fmt.Fprintf(&b, "Вентилятор: %s (%s) — %d об/мин\n", orDashStr(label), chip, rpm)
					fans++
				}
			}
		}
		vFiles, _ := filepath.Glob(dir + "/in*_input")
		for _, f := range vFiles {
			if v := hwmonFile(dir, filepath.Base(f)); v != "" {
				if mv, err := strconv.ParseFloat(v, 64); err == nil && mv > 0 {
					label := hwmonFile(dir, strings.TrimSuffix(filepath.Base(f), "_input")+"_label")
					fmt.Fprintf(&b, "Напряжение: %s (%s) — %.2f В\n", orDashStr(label), chip, mv/1e6)
					volts++
				}
			}
		}
	}
	if fans == 0 && volts == 0 {
		return "Вентиляторы и напряжения: данные hwmon недоступны.\nУстановите lm-sensors и выполните 'sudo sensors-detect' — данные появятся в /sys/class/hwmon."

type lhmSensor struct {
	Name       string  `json:"Name"`
	SensorType string  `json:"SensorType"`
	Value      float64 `json:"Value"`
}

type sensorReading struct {
	name  string
	value float64
}

// querySensorsOnce опрашивает оба известных пространства имён
// и возвращает прочитанные сенсоры и имя сработавшего пространства.
func querySensorsOnce() (fans []sensorReading, volts []sensorReading, namespace string) {
	namespaces := []string{"root/LibreHardwareMonitor", "root/OpenHardwareMonitor"}

	for _, ns := range namespaces {
		out, err := runSensorsPS(ns)
		if err != nil {
			continue // пространства имён нет — пробуем следующее
		}

		trimmed := strings.TrimSpace(out)
		if trimmed == "" || trimmed == "[]" {
			continue // пространство есть, но подходящих сенсоров нет
		}

		var raw []lhmSensor
		if err := json.Unmarshal([]byte(trimmed), &raw); err != nil {
			continue
		}

		for _, s := range raw {
			switch s.SensorType {
			case "Fan":
				fans = append(fans, sensorReading{name: s.Name, value: s.Value})
			case "Voltage":
				// нулевые каналы напряжения — почти всегда неиспользуемые
				// разъёмы материнской платы, не захламляем отчёт
				if s.Value != 0 {
					volts = append(volts, sensorReading{name: s.Name, value: s.Value})
				}
			}
		}
		return fans, volts, ns
	}

	return nil, nil, ""
}

// runSensorsPS делает один WMI-запрос через PowerShell.
// $ErrorActionPreference='Stop' превращает отсутствие пространства имён
// в ошибку, которую querySensorsOnce перехватывает и переключается на резерв.
func runSensorsPS(namespace string) (string, error) {
	ps := `$ErrorActionPreference = 'Stop'; ` +
		`$s = Get-CimInstance -Namespace ` + namespace + ` -ClassName Sensor | ` +
		`Where-Object { $_.SensorType -in 'Fan','Voltage' -and $null -ne $_.Value }; ` +
		`@($s | Select-Object Name, SensorType, Value) | ConvertTo-Json -Compress`

	full := `[Console]::OutputEncoding=[Text.Encoding]::UTF8; ` + ps
	out, err := exec.Command("powershell", "-NoProfile", "-Command", full).Output()
	return string(out), err
}

		typeData, err := os.ReadFile(typePath)

		if err != nil {
			continue
	}
	tempC /= 1000 // конвертируем из миллиградусов

	}

	}

	outStr := strings.TrimSpace(string(out))

		result.Value = "Датчики температуры не обнаружены.\nПримечание: метод MSAcpi_ThermalZoneTemperature недоступен на этой системе."
		return result
	}

	// WMI может вернуть несколько значений — построчно
	lines := strings.Split(outStr, "\n")
	var sb strings.Builder
	found := false

	for _, line := range lines {
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}

			continue
		}

		sb.WriteString(fmt.Sprintf("Температура (Thermal Zone): %.1f°C\n", tempC))
		found = true
	}

	if !found {
		sb.WriteString("Датчики CPU не распознаны этим методом.\n")

	result.Value = sb.String()

		utils.Println(utils.Dim("Температура CPU: недоступна (требуются права админа/драйверы)"))
		utils.Println("")
	}

	// --- СТАТУС СИСТЕМЫ ---
	stRAM := thresholds.ForRAM(vm.UsedPercent)

	stCPU := thresholds.StatusUnknown
	if len(totalCpu) > 0 {
		stCPU = thresholds.ForCPU(totalCpu[0])
	}

	stTemp := thresholds.ForTemp(temp)
	overall := thresholds.Worst(stCPU, stRAM, stTemp)

	utils.Println(utils.Header("=== СОСТОЯНИЕ СИСТЕМЫ ==="))
	utils.Printf("CPU: %s | RAM: %s | Температура: %s | Общий статус: %s",
		statusColored(stCPU), statusColored(stRAM), statusColored(stTemp), statusColored(overall))
	utils.Println("")


	// Подсказка внизу
	utils.Println(utils.Header(strings.Repeat("=", 40)))
	utils.Println(utils.Bold("[Enter]") + utils.Dim(" - Выход в меню"))
}

// isEnterPressed проверяет, была ли нажата клавиша Enter (неблокирующее чтение)
// В raw-режиме Enter = \r (0x0D) на Windows и Linux
func isEnterPressed() bool {
	buffer := make([]byte, 1)

	os.Stdin.SetReadDeadline(time.Now())
	n, err := os.Stdin.Read(buffer)

	if n > 0 && err == nil {
		// \r = 0x0D — Enter в raw-режиме
		return buffer[0] == '\r' || buffer[0] == '\n'
	}
	return false
}

func getCPUTempWindows() float64 {

	RAMUsedPct  float64   `json:"ram_used_pct"`

	SwapTotalGB float64   `json:"swap_total_gb"`

}

// Кэш температуры — обновляется в фоне каждые 5 секунд
var (
	cachedTemp     float64
	tempMu         sync.Mutex
	tempInitialized bool
)

func init() {
	// Фоновое обновление температуры
	go func() {
		for {
			t := getCPUTempWindows()
			tempMu.Lock()
			cachedTemp = t
			tempInitialized = true
			tempMu.Unlock()
			time.Sleep(5 * time.Second)
		}
	}()
}

// GetStats собирает текущие метрики системы
func GetStats() Stats {
	stats := Stats{
		Timestamp: time.Now().Format("15:04:05"),
	}

	vm, _ := mem.VirtualMemory()
	if vm != nil {
		stats.RAMUsedPct = vm.UsedPercent
		stats.RAMUsedGB = float64(vm.Used) / 1024 / 1024 / 1024
		stats.RAMTotalGB = float64(vm.Total) / 1024 / 1024 / 1024
	}

	swap, _ := mem.SwapMemory()
	if swap != nil {
		stats.SwapUsedPct = swap.UsedPercent
		stats.SwapUsedGB = float64(swap.Used) / 1024 / 1024 / 1024
		stats.SwapTotalGB = float64(swap.Total) / 1024 / 1024 / 1024
	}

	cpuCores, _ := cpu.Percent(0, true)
	stats.CPUCores = cpuCores

	cpuTotal, _ := cpu.Percent(0, false)
	if len(cpuTotal) > 0 {
		stats.CPUTotal = cpuTotal[0]
	}

	// Берём температуру из кэша — мгновенно, без задержки
	tempMu.Lock()
	stats.CPUTemp = cachedTemp
	tempMu.Unlock()

	return stats
}

// statusRu переводит статус в русское слово для вывода.
func statusRu(s thresholds.Status) string {
	switch s {
	case thresholds.StatusNormal:
		return "НОРМА"
	case thresholds.StatusWarning:
		return "ВНИМАНИЕ"
	case thresholds.StatusCritical:
		return "КРИТИЧНО"
	}
	return "НЕТ ДАННЫХ"
}

// statusColored возвращает русское название статуса, окрашенное ANSI-цветом
// (зелёный/жёлтый/красный/серый). В веб-отчёте цветовые коды вычищаются
// регуляркой при выводе результата.
func statusColored(s thresholds.Status) string {
	colors := map[thresholds.Status]string{
		thresholds.StatusNormal:   "\033[32m",
		thresholds.StatusWarning:  "\033[33m",
		thresholds.StatusCritical: "\033[31m",
		thresholds.StatusUnknown:  "\033[90m",
	}
	return colors[s] + statusRu(s) + "\033[0m"
}

// tempStr форматирует температуру: -1 означает «нет данных».
func tempStr(t float64) string {
	if t < 0 {
		return "н/д"
	}
	return fmt.Sprintf("%.1f°C", t)
}

// cpuTemperature возвращает температуру CPU (°C) или -1, если недоступна.
func cpuTemperature() float64 {
	if runtime.GOOS == "windows" {
		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		out, err := exec.CommandContext(ctx, "powershell", "-NoProfile", "-Command",
			"(Get-CimInstance -Namespace root/wmi -ClassName MSAcpi_ThermalZoneTemperature | Select-Object -First 1).CurrentTemperature").Output()
		if err != nil {
			return -1
		}
		v, err := strconv.ParseFloat(strings.TrimSpace(string(out)), 64)
		if err != nil {
			return -1
		}
		return v/10 - 273.15
	}
	entries, err := os.ReadDir("/sys/class/thermal")
	if err != nil {
		return -1
	}
	best := -1.0
	for _, e := range entries {
		if !strings.HasPrefix(e.Name(), "thermal_zone") {
			continue
		}
		raw, err := os.ReadFile("/sys/class/thermal/" + e.Name() + "/temp")
		if err != nil {
			continue
		}
		v, err := strconv.ParseFloat(strings.TrimSpace(string(raw)), 64)
		if err != nil {
			continue
		}
		if c := v / 1000; c > best {
			best = c
		}
	}
	return best
}

// stats — статистика за время теста.
type stats struct {
	maxTemp  float64
	sumTemp  float64
	nTemp    int
	overheat bool
	lastLoad float64 // последняя измеренная загрузка CPU, %
	hasLoad  bool    // была ли хотя бы одна успешная выборка загрузки
}

func (s *stats) avg() float64 {
	if s.nTemp == 0 {
		return -1
	}
	return s.sumTemp / float64(s.nTemp)
}

// monitorLoop раз в секунду печатает загрузку, температуру и статус,
// при превышении предела отменяет тест.
func monitorLoop(ctx context.Context, maxTemp float64, st *stats, cancel context.CancelFunc, done chan<- struct{}) {
	defer close(done)
	st.maxTemp = -1
	start := time.Now()
	t := time.NewTicker(time.Second)
	defer t.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case <-t.C:
			temp := cpuTemperature()
			load := -1.0
			if p, err := cpu.Percent(0, false); err == nil && len(p) > 0 {
				load = p[0]
			}
			if temp > 0 {
				st.sumTemp += temp
				st.nTemp++
				if temp > st.maxTemp {
					st.maxTemp = temp
				}
			}
			if load >= 0 {
				st.lastLoad = load
				st.hasLoad = true
				stCPU := thresholds.ForCPU(load)
				fmt.Printf("\r  [%02ds] CPU %.0f%%  темп. %s  CPU: %s   ",
					int(time.Since(start).Seconds()), load, tempStr(temp), statusColored(stCPU))
			} else {
				fmt.Printf("\r  [%02ds] темп. %s   ",
					int(time.Since(start).Seconds()), tempStr(temp))
			}
			if maxTemp > 0 && temp >= maxTemp {
				st.overheat = true
				fmt.Println("\n  !! ПРЕДЕЛ ТЕМПЕРАТУРЫ ДОСТИГНУТ — тест остановлен")
				cancel()
				return
			}
		}
	}
}

// cpuLoadAll грузит заданное число ядер вычислениями до отмены контекста.
func cpuLoadAll(ctx context.Context, cores int, ops *uint64) {
	var wg sync.WaitGroup
	for i := 0; i < cores; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			x := 0.5
			for {
				select {
				case <-ctx.Done():
					return
				default:
				}
				x = math.Sqrt(x*1.0000001 + 0.0000001)
				x = math.Mod(x*3.7, 2.0)
				_ = math.Sin(x) * math.Cos(x)
				atomic.AddUint64(ops, 1)
			}
		}()
	}
	wg.Wait()
}

// RunCPU — стресс-тест процессора: все ядра, живая строка прогресса,
// контроль температуры, сводка в конце.
func RunCPU(durationSec int, maxTemp float64) string {
	if runtime.GOOS != "windows" && runtime.GOOS != "linux" {
		return "Стресс-тест CPU: поддерживается на Windows и Linux"
	}
	cores := runtime.NumCPU()
	if maxTemp <= 0 {
		maxTemp = 90
	}
	dur := time.Duration(durationSec) * time.Second
	if dur < 5*time.Second {
		dur = 5 * time.Second
	}

	fmt.Printf("Стресс-тест CPU: %d ядер, %v (аварийная остановка при %s)\n",
		cores, dur, tempStr(maxTemp))
	before := cpuTemperature()

	ctx, cancel := context.WithTimeout(context.Background(), dur)
	defer cancel()

	var st stats
	var ops uint64
	monDone := make(chan struct{})
	go monitorLoop(ctx, maxTemp, &st, cancel, monDone)

	start := time.Now()
	cpuLoadAll(ctx, cores, &ops)
	elapsed := time.Since(start)
	cancel()  // гарантируем остановку монитора
	<-monDone // ждём, пока монитор напечатает свою последнюю строку
	fmt.Println()

	perCore := float64(atomic.LoadUint64(&ops)) / elapsed.Seconds() / float64(cores) / 1e6

	// Статус по данным, измеренным под нагрузкой (последняя выборка монитора),
	// а не по «остывшей» системе после остановки горутин.
	stCPU := thresholds.StatusUnknown
	if st.hasLoad {
		stCPU = thresholds.ForCPU(st.lastLoad)
	}
	overall := thresholds.Worst(stCPU, thresholds.ForTemp(st.maxTemp))

	var b strings.Builder
	b.WriteString("\nРезультаты стресс-теста CPU:\n")
	b.WriteString(fmt.Sprintf("  Ядер нагружено:        %d\n", cores))
	b.WriteString(fmt.Sprintf("  Длительность:          %v\n", elapsed.Round(time.Second)))
	b.WriteString(fmt.Sprintf("  Производительность:    %.1f млн операций/с на ядро\n", perCore))
	b.WriteString(fmt.Sprintf("  Температура до теста:  %s\n", tempStr(before)))
	b.WriteString(fmt.Sprintf("  Под нагрузкой:         средняя %s, макс. %s\n",
		tempStr(st.avg()), tempStr(st.maxTemp)))
	b.WriteString(fmt.Sprintf("  Состояние системы:     %s\n", statusColored(overall)))
	if st.overheat {
		b.WriteString("  ВНИМАНИЕ: тест остановлен по пределу температуры!\n")
	}
	return b.String()
}

// u64Equal сравнивает два среза uint64 поэлементно.
func u64Equal(a, b []uint64) bool {
	if len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}

// RunRAM — стресс-тест оперативной памяти: многопоточная запись блоками по 8 байт,
// проверка целостности данных, защита от исчерпания свободной памяти.
func RunRAM(durationSec int) string {
	if runtime.GOOS != "windows" && runtime.GOOS != "linux" {
		return "Стресс-тест RAM: поддерживается на Windows и Linux"
	}
	dur := time.Duration(durationSec) * time.Second
	if dur < 5*time.Second {
		dur = 5 * time.Second
	}

	vm, err := mem.VirtualMemory()
	if err != nil {
		return fmt.Sprintf("Стресс-тест RAM: не удалось получить данные о памяти: %v", err)
	}
	fmt.Printf("Стресс-тест RAM: длительность %v, свободно %.1f ГБ из %.1f ГБ\n",
		dur, float64(vm.Available)/1e9, float64(vm.Total)/1e9)

	var lowMem int32 // 1 — памяти стало критически мало
	var maxUsedPct float64
	ctx, cancel := context.WithTimeout(context.Background(), dur)
	defer cancel()

	monDone := make(chan struct{})
	go func() {
		defer close(monDone)
		start := time.Now()
		t := time.NewTicker(time.Second)
		defer t.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-t.C:
				v, err := mem.VirtualMemory()
				if err != nil {
					continue
				}
				usedPct := 0.0
				if v.Total > 0 {
					usedPct = (1 - float64(v.Available)/float64(v.Total)) * 100
					if usedPct > maxUsedPct {
						maxUsedPct = usedPct
					}
				}
				fmt.Printf("\r  [%02ds] доступно: %.1f ГБ  RAM: %s   ",
					int(time.Since(start).Seconds()), float64(v.Available)/1e9,
					statusColored(thresholds.ForRAM(usedPct)))
				if v.Available < 500*1024*1024 {
					atomic.StoreInt32(&lowMem, 1)
					return
				}
			}
		}
	}()

	const chunkMB = 128
	buf := make([]uint64, chunkMB*1024*1024/8) // 128 МБ как массив 8-байтовых ячеек
	workers := runtime.NumCPU()
	segment := len(buf) / workers
	if segment < 1024 {
		segment = len(buf)
		workers = 1
	}
	const checkPerSegment = 512 // сколько ячеек проверяем в каждой части

	// Заполняет буфер по частям: каждый поток пишет свой сегмент.
	fillPattern := func() {
		var wg sync.WaitGroup
		for w := 0; w < workers; w++ {
			w := w
			wg.Add(1)
			go func() {
				defer wg.Done()
				start := w * segment
				end := start + segment
				if w == workers-1 {
					end = len(buf)
				}
				for i := start; i < end; i++ {
					buf[i] = uint64(i)*0x9E3779B97F4A7C15 + 12345
				}
			}()
		}
		wg.Wait()
	}

	// Эталонные значения для проверки целостности: начало каждого сегмента.
	fillPattern()
	reference := make([]uint64, 0, workers*checkPerSegment)
	for w := 0; w < workers; w++ {
		reference = append(reference, buf[w*segment:w*segment+checkPerSegment]...)
	}

	start := time.Now()
	cycles := 0
loop:
	for {
		select {
		case <-ctx.Done():
			break loop
		default:
		}
		if atomic.LoadInt32(&lowMem) == 1 {
			break loop
		}
		fillPattern() // запись 128 МБ всеми потоками
		ok := true
		for w := 0; w < workers && ok; w++ {
			seg := buf[w*segment : w*segment+checkPerSegment]
			ref := reference[w*checkPerSegment : (w+1)*checkPerSegment]
			if !u64Equal(seg, ref) {
				ok = false
			}
		}
		if !ok {
			break loop
		}
		cycles++
	}
	elapsed := time.Since(start)
	cancel()
	<-monDone
	fmt.Println()

	vmAfter, err := mem.VirtualMemory()
	afterStr := "н/д"
	if err == nil {
		afterStr = fmt.Sprintf("%.1f ГБ свободно", float64(vmAfter.Available)/1e9)
	}
	overall := thresholds.StatusUnknown
	if maxUsedPct > 0 {
		overall = thresholds.ForRAM(maxUsedPct)
	}
	bw := float64(cycles) * chunkMB / elapsed.Seconds() // МБ/с записи

	var b strings.Builder
	b.WriteString("\nРезультаты стресс-теста RAM:\n")
	b.WriteString(fmt.Sprintf("  Длительность:        %v\n", elapsed.Round(time.Second)))
	b.WriteString(fmt.Sprintf("  Потоков записи:      %d (блоками по 8 байт)\n", workers))
	b.WriteString(fmt.Sprintf("  Полных циклов:       %d (запись 128 МБ + проверка)\n", cycles))
	b.WriteString(fmt.Sprintf("  Пропускная способность записи: %.0f МБ/с\n", bw))
	b.WriteString(fmt.Sprintf("  Свободная память до:   %.1f ГБ из %.1f ГБ\n", float64(vm.Available)/1e9, float64(vm.Total)/1e9))
	b.WriteString(fmt.Sprintf("  Свободная память после: %s\n", afterStr))
	b.WriteString(fmt.Sprintf("  Состояние системы:     %s\n", statusColored(overall)))
	if atomic.LoadInt32(&lowMem) == 1 {
		b.WriteString("  ВНИМАНИЕ: тест остановлен — свободной памяти стало слишком мало.\n")
	}
	return b.String()
}

// RunDisk — стресс-тест диска: запись файла-жертвы блоками,
// живой счётчик скорости, средняя скорость за тест, автопомойка.
func RunDisk(durationSec int) string {
	if runtime.GOOS != "windows" && runtime.GOOS != "linux" {
		return "Стресс-тест диска: поддерживается на Windows и Linux"
	}
	dur := time.Duration(durationSec) * time.Second
	if dur < 5*time.Second {
		dur = 5 * time.Second
	}

	path := filepath.Join(os.TempDir(), "pc-diagnostic-stress.tmp")
	fmt.Printf("Стресс-тест диска: длительность %v, файл: %s\n", dur, path)

	f, err := os.Create(path)
	if err != nil {
		return fmt.Sprintf("Стресс-тест диска: не удалось создать файл: %v", err)
	}
	// Уборка гарантирована при любом исходе: файл закрывается и удаляется.
	defer func() {
		f.Close()
		os.Remove(path)
	}()

	const chunkMB = 4
	chunk := make([]byte, chunkMB*1024*1024) // 4 МБ
	for i := range chunk {
		chunk[i] = byte(i * 31)
	}

	var totalWritten uint64 // МБ

	ctx, cancel := context.WithTimeout(context.Background(), dur)
	defer cancel()

	monDone := make(chan struct{})
	go func() {
		defer close(monDone)
		tick := time.NewTicker(time.Second)
		defer tick.Stop()
		var lastBytes uint64
		for {
			select {
			case <-ctx.Done():
				return
			case <-tick.C:
				b := atomic.LoadUint64(&totalWritten)
				fmt.Printf("\r  скорость: %6.0f МБ/с   ", float64(b-lastBytes))
				lastBytes = b
			}
		}
	}()

	start := time.Now()
loop:
	for {
		select {
		case <-ctx.Done():
			break loop
		default:
		}
		if _, err := f.Write(chunk); err != nil {
			cancel()
			<-monDone
			fmt.Println()
			return fmt.Sprintf("Стресс-тест диска: ошибка записи: %v", err)
		}
		atomic.AddUint64(&totalWritten, chunkMB)
	}
	elapsed := time.Since(start)
	cancel()
	<-monDone
	fmt.Println()

	totalMB := atomic.LoadUint64(&totalWritten)
	bw := float64(totalMB) / elapsed.Seconds()

	var b strings.Builder
	b.WriteString("\nРезультаты стресс-теста диска:\n")
	b.WriteString(fmt.Sprintf("  Длительность:         %v\n", elapsed.Round(time.Second)))
	b.WriteString(fmt.Sprintf("  Записано:             %.1f ГБ\n", float64(totalMB)/1024))
	b.WriteString(fmt.Sprintf("  Средняя скорость записи: %.0f МБ/с\n", bw))
	b.WriteString("  Файл-жертва удалён.\n")
	return b.String()
}

// RunCombined — комбинированный стресс-тест: нагрузка на все ядра CPU
// и запись RAM работают одновременно. Показывает, как компоненты
// влияют друг на друга при совместной нагрузке.
func RunCombined(durationSec int, maxTemp float64) string {
	if runtime.GOOS != "windows" && runtime.GOOS != "linux" {
		return "Комбинированный тест: поддерживается на Windows и Linux"
	}
	cores := runtime.NumCPU()
	if maxTemp <= 0 {
		maxTemp = 90
	}
	dur := time.Duration(durationSec) * time.Second
	if dur < 5*time.Second {
		dur = 5 * time.Second
	}

	vm0, err := mem.VirtualMemory()
	if err != nil {
		return fmt.Sprintf("Комбинированный тест: не удалось получить данные о памяти: %v", err)
	}
	fmt.Printf("Комбинированный стресс-тест: %d ядер CPU + запись RAM, %v (аварийная остановка при %s)\n",
		cores, dur, tempStr(maxTemp))
	fmt.Printf("Свободно памяти: %.1f ГБ из %.1f ГБ\n",
		float64(vm0.Available)/1e9, float64(vm0.Total)/1e9)

	before := cpuTemperature()

	ctx, cancel := context.WithTimeout(context.Background(), dur)
	defer cancel()

	var cpuOps uint64
	var ramMB uint64
	var ramCycles uint64
	var overheat int32

	// Нагрузка CPU: все ядра, как в RunCPU.
	go cpuLoadAll(ctx, cores, &cpuOps)

	// Нагрузка RAM: потоки пишут и проверяют буфер 128 МБ в цикле, как в RunRAM.
	ramDone := make(chan struct{})
	go func() {
		defer close(ramDone)
		const chunkMB = 128
		buf := make([]uint64, chunkMB*1024*1024/8)
		workers := runtime.NumCPU()
		segment := len(buf) / workers
		if segment < 1024 {
			segment = len(buf)
			workers = 1
		}
		fillOnce := func() {
			var wg sync.WaitGroup
			for w := 0; w < workers; w++ {
				w := w
				wg.Add(1)
				go func() {
					defer wg.Done()
					lo := w * segment
					hi := lo + segment
					if w == workers-1 {
						hi = len(buf)
					}
					for i := lo; i < hi; i++ {
						buf[i] = uint64(i)*0x9E3779B97F4A7C15 + 12345
					}
				}()
			}
			wg.Wait()
		}
		fillOnce()
		ref := make([]uint64, 512)
		copy(ref, buf[:512])
		for {
			select {
			case <-ctx.Done():
				return
			default:
			}
			fillOnce()
			if !u64Equal(buf[:512], ref) {
				return // целостность данных нарушена — стоп
			}
			atomic.AddUint64(&ramMB, chunkMB)
			atomic.AddUint64(&ramCycles, 1)
		}
	}()

	// Монитор: раз в секунду — загрузка CPU, скорость RAM, память, температура.
	start := time.Now()
	monDone := make(chan struct{})
	var maxSeen float64
	var maxRamPct float64
	var tempSum float64
	var tempN int
	var cpuSum float64
	var cpuN int
	go func() {
		defer close(monDone)
		tick := time.NewTicker(time.Second)
		defer tick.Stop()
		var lastRam uint64
		for {
			select {
			case <-ctx.Done():
				return
			case <-tick.C:
				ram := atomic.LoadUint64(&ramMB)
				ramSpeed := float64(ram - lastRam)
				lastRam = ram
				pcts, _ := cpu.Percent(0, true)
				cpuAvg := 0.0
				for _, p := range pcts {
					cpuAvg += p
				}
				if len(pcts) > 0 {
					cpuAvg /= float64(len(pcts))
					cpuSum += cpuAvg
					cpuN++
				}
				temp := cpuTemperature()
				if temp > 0 {
					tempSum += temp
					tempN++
					if temp > maxSeen {
						maxSeen = temp
					}
				}
				vm, _ := mem.VirtualMemory()
				avail := 0.0
				if vm != nil {
					avail = float64(vm.Available) / 1e9
					if vm.Total > 0 {
						usedPct := (1 - float64(vm.Available)/float64(vm.Total)) * 100
						if usedPct > maxRamPct {
							maxRamPct = usedPct
						}
					}
				}
				stCPU := thresholds.ForCPU(cpuAvg)
				stRAM := thresholds.ForRAM(maxRamPct)
				fmt.Printf("\r  [%02ds] CPU %3.0f%%  RAM: %4.0f МБ/с  своб. %.1f ГБ  темп %s  %s   ",
					int(time.Since(start).Seconds()), cpuAvg, ramSpeed, avail,
					tempStr(temp), statusColored(thresholds.Worst(stCPU, stRAM)))
				if temp >= maxTemp && temp > 0 {
					atomic.StoreInt32(&overheat, 1)
					fmt.Printf("\r  [%02ds] ПРЕВЫШЕН ПРЕДЕЛ ТЕМПЕРАТУРЫ (%s), остановка   \n",
						int(time.Since(start).Seconds()), tempStr(temp))
					cancel()
					return
				}
			}
		}
	}()

	// Ждём окончания нагрузки (RAM-горутина завершается по таймауту
	// или при нарушении целостности).
	<-ramDone
	elapsed := time.Since(start)
	cancel()
	<-monDone
	fmt.Println()

	perCore := float64(atomic.LoadUint64(&cpuOps)) / elapsed.Seconds() / float64(cores) / 1e6
	ramBw := float64(atomic.LoadUint64(&ramMB)) / elapsed.Seconds()

	// Статус по данным, измеренным под нагрузкой.
	stCPU := thresholds.StatusUnknown
	if cpuN > 0 {
		stCPU = thresholds.ForCPU(cpuSum / float64(cpuN))
	}
	stRAM := thresholds.StatusUnknown
	if maxRamPct > 0 {
		stRAM = thresholds.ForRAM(maxRamPct)
	}
	stTemp := thresholds.StatusUnknown
	if tempN > 0 && maxSeen > 0 {
		stTemp = thresholds.ForTemp(maxSeen)
	}
	overall := thresholds.Worst(stCPU, stRAM, stTemp)

	var b strings.Builder
	b.WriteString("\nРезультаты комбинированного стресс-теста:\n")
	b.WriteString(fmt.Sprintf("  Длительность:           %v\n", elapsed.Round(time.Second)))
	b.WriteString(fmt.Sprintf("  CPU (%d ядер):           %.1f млн операций/с на ядро\n", cores, perCore))
	b.WriteString(fmt.Sprintf("  RAM (запись):           %.0f МБ/с\n", ramBw))
	b.WriteString(fmt.Sprintf("  Полных циклов RAM:      %d\n", atomic.LoadUint64(&ramCycles)))
	if cpuN > 0 {
		b.WriteString(fmt.Sprintf("  Загрузка CPU средняя:   %.0f%%\n", cpuSum/float64(cpuN)))
	}
	b.WriteString(fmt.Sprintf("  Температура до теста:   %s\n", tempStr(before)))
	if tempN > 0 {
		b.WriteString(fmt.Sprintf("  Под нагрузкой:          средняя %s, макс. %s\n",
			tempStr(tempSum/float64(tempN)), tempStr(maxSeen)))
	}
	vmAfter, err := mem.VirtualMemory()
	if err == nil {
		b.WriteString(fmt.Sprintf("  Свободная память после: %.1f ГБ из %.1f ГБ\n",
			float64(vmAfter.Available)/1e9, float64(vmAfter.Total)/1e9))
	}
	b.WriteString(fmt.Sprintf("  Состояние системы:      %s\n", statusColored(overall)))
	if atomic.LoadInt32(&overheat) == 1 {
		b.WriteString("  ВНИМАНИЕ: тест остановлен по пределу температуры!\n")
	}
	return b.String()
}

// BuildFromHardware строит секции из результатов hardware.RunDiagnostics()
func BuildFromHardware(results []hardware.DiagnosticResult) Report {
	r := Report{
		Title: "Отчёт диагностики ПК",
		Date:  time.Now().Format("02.01.2006 15:04:05"),
	}

	for _, res := range results {
		sec := Section{Title: res.Name}
		if res.Error != nil {
			sec.Rows = append(sec.Rows, Row{Label: "Ошибка", Value: res.Error.Error()})
		} else {
			sec.Rows = append(sec.Rows, parseTextBlock(res.Value)...)
		}
		r.Sections = append(r.Sections, sec)
	}
	return r
}

// AddText добавляет текстовый блок (например, из system.Get*Info())
func (r *Report) AddText(title, text string) {
	r.Sections = append(r.Sections, Section{
		Title: title,
		Rows:  parseTextBlock(text),
	})
}

// parseTextBlock превращает текст в пары «метка: значение».
// Строки без «:» сохраняются как безымянные строки значения.
func parseTextBlock(text string) []Row {
	var rows []Row
	for _, line := range strings.Split(text, "\n") {
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}
		if idx := strings.Index(line, ":"); idx > 0 {
			rows = append(rows, Row{
				Label: strings.TrimSpace(line[:idx]),
				Value: strings.TrimSpace(line[idx+1:]),
			})
		} else {
			rows = append(rows, Row{Value: line})
		}
	}
	return rows
}

type Reporter interface {
	Write(r Report) ([]byte, error)
	Ext() string
}
// ---------- TXT ----------

type TXTReporter struct{}

func (TXTReporter) Ext() string { return "txt" }

func (TXTReporter) Write(r Report) ([]byte, error) {
	var b strings.Builder

	b.WriteString(r.Title + "\n")
	b.WriteString("Дата: " + r.Date + "\n\n")

	for _, sec := range r.Sections {
		b.WriteString(sec.Title + ":\n")
		for _, row := range sec.Rows {
			if row.Label != "" {
				b.WriteString(" " + row.Label + ": " + row.Value + "\n")
			} else {
				b.WriteString(" " + row.Value + "\n")
			}
		}
		b.WriteString("\n---\n\n")
	}

	return []byte(b.String()), nil
}
// ---------- JSON ----------

type JSONReporter struct{}

func (JSONReporter) Ext() string { return "json" }

func (JSONReporter) Write(r Report) ([]byte, error) {
	return json.MarshalIndent(r, "", "  ")
}

// ---------- CSV ----------

type CSVReporter struct{}

func (CSVReporter) Ext() string { return "csv" }

func (CSVReporter) Write(r Report) ([]byte, error) {
	var buf bytes.Buffer

	// BOM, чтобы Excel корректно открыл кириллицу
	buf.WriteString("\uFEFF")

	w := csv.NewWriter(&buf)
	w.Comma = ';' // Excel по умолчанию понимает точку с запятой

	w.Write([]string{r.Title})
	w.Write([]string{"Дата", r.Date})
	w.Write([]string{})

	for _, sec := range r.Sections {
		w.Write([]string{"## " + sec.Title})
		for _, row := range sec.Rows {
			w.Write([]string{sec.Title, row.Label, row.Value})
		}
		w.Write([]string{})
	}
	w.Flush()
	if err := w.Error(); err != nil {
		return nil, fmt.Errorf("запись CSV: %w", err)
	}
	return buf.Bytes(), nil
}

func GetAudioInfo() string {

	}

		devices, err = getAudioViaPowerShell()
		if err == nil && len(devices) > 0 {

			for _, device := range devices {

// +build linux
func getAudioFromProc() (string, error) {
	var result strings.Builder
	var hasData bool

	cardsData, err := readFileSafely("/proc/asound/cards")
	if err == nil {
		result.WriteString("Звуковые карты (/proc/asound/cards):\n")
		cardLines := strings.Split(cardsData, "\n")
		for _, line := range cardLines {
			trimmed := strings.TrimSpace(line)
			if trimmed != "" {
				result.WriteString("  " + trimmed + "\n")
			hasData = true
		}
	}
	}

	pcmData, err := readFileSafely("/proc/asound/pcm")
	if err == nil {
		if hasData {
			result.WriteString("\n") // разделитель между секциями
		}
		result.WriteString("PCM устройства (/proc/asound/pcm):\n")
		pcmLines := strings.Split(pcmData, "\n")
		for _, line := range pcmLines {
			trimmed := strings.TrimSpace(line)
			if trimmed != "" {
				result.WriteString("  " + trimmed + "\n")
			hasData = true
		}
	}
	}

	if !hasData {
		return "", fmt.Errorf("нет данных в /proc/asound/")
	}

	return result.String(), nil
}

// +build windows
func getAudioViaWMI() ([]string, error) {
	err := ole.CoInitialize(0)
	if err != nil {
		return nil, err
	}
	defer ole.CoUninitialize()

	unknown, err := oleutil.CreateObject("WbemScripting.SWbemLocator")
	if err != nil {
		return nil, err
	}
	defer unknown.Release()

	locator, err := unknown.QueryInterface(ole.IID_IDispatch)
	if err != nil {
		return nil, err
	}
	defer locator.Release()

	serviceRaw, err := oleutil.CallMethod(locator, "ConnectServer", ".", "root\\CIMV2")
	if err != nil {
		return nil, fmt.Errorf("не удалось подключиться к WMI: %v", err)
	}

	service := serviceRaw.ToIDispatch()
	defer service.Release()

	query := "SELECT Name, Manufacturer, DeviceID, Status FROM Win32_SoundDevice"
	resultsRaw, err := oleutil.CallMethod(service, "ExecQuery", query)
	if err != nil {
		return nil, fmt.Errorf("WMI запрос не выполнен: %v", err)
	}

	results := resultsRaw.ToIDispatch()
	defer results.Release()

	countVar, err := oleutil.GetProperty(results, "Count")
	if err != nil || countVar.Val == 0 {
		return nil, fmt.Errorf("звуковые устройства не найдены через WMI")
	}

	var devices []string
	for i := 0; i < int(countVar.Val); i++ {
		itemRaw, err := oleutil.CallMethod(results, "ItemIndex", i)
		if err != nil {
			continue
	}
	item := itemRaw.ToIDispatch()

	nameRaw, err := oleutil.GetProperty(item, "Name")
	name := "Неизвестно"
	if err == nil {
		name = nameRaw.ToString()
	}

	manufacturerRaw, err := oleutil.GetProperty(item, "Manufacturer")
	manufacturer := "Неизвестно"
	if err == nil {
		manufacturer = manufacturerRaw.ToString()
	}

	deviceIDRaw, err := oleutil.GetProperty(item, "DeviceID")
	deviceID := "Неизвестно"
	if err == nil {
		deviceID = deviceIDRaw.ToString()
	}

	statusRaw, err := oleutil.GetProperty(item, "Status")
	status := "Неизвестно"
	if err == nil {
		status = statusRaw.ToString()
	}

	devices = append(devices, fmt.Sprintf("%s (%s, ID: %s, Статус: %s)",
		name, manufacturer, deviceID, status))

	item.Release() // явное освобождение ресурсов
	}

	return devices, nil
}

package system

import (
	"bufio"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
)

// GetAutostartInfo — программы в автозагрузке (Linux):
// .desktop-файлы, юниты systemd пользователя, cron.
func GetAutostartInfo() string {
	home, _ := os.UserHomeDir()
	var b strings.Builder
	count := 0

	for _, dir := range []string{filepath.Join(home, ".config/autostart"), "/etc/xdg/autostart"} {
		files, _ := filepath.Glob(filepath.Join(dir, "*.desktop"))
		for _, f := range files {
			name, execLine, hidden := "", "", false
			fh, err := os.Open(f)
			if err != nil {
				continue
			}
			sc := bufio.NewScanner(fh)
			for sc.Scan() {
				line := strings.TrimSpace(sc.Text())
				switch {
				case strings.HasPrefix(line, "Name="):
					name = strings.TrimPrefix(line, "Name=")
				case strings.HasPrefix(line, "Exec="):
					execLine = strings.TrimPrefix(line, "Exec=")
				case line == "Hidden=true" || line == "NoDisplay=true":
					hidden = true
				}
			}
			fh.Close()
			if hidden {
				continue
			}
			count++
			fmt.Fprintf(&b, "%d. %s — %s\n", count, orDash(name), orDash(execLine))
		}
	}

	units, _ := filepath.Glob(filepath.Join(home, ".config/systemd/user/*.service"))
	for _, u := range units {
		count++
		fmt.Fprintf(&b, "%d. systemd-user: %s\n", count, filepath.Base(u))
	}

	if out, err := exec.Command("crontab", "-l").Output(); err == nil && strings.TrimSpace(string(out)) != "" {
		for _, line := range strings.Split(strings.TrimSpace(string(out)), "\n") {
			if line != "" && !strings.HasPrefix(line, "#") {
				count++
				fmt.Fprintf(&b, "%d. cron: %s\n", count, line)
			}
		}
	}

	if count == 0 {
		return "Автозагрузка: записи не найдены"
	}
	return b.String()
}

	result.WriteString("=== ПРОВЕРКА ЦЕЛОСТНОСТИ СИСТЕМНЫХ ФАЙЛОВ ===\n")

	switch runtime.GOOS {

		raw, err := cmd.CombinedOutput()
		if err != nil {
			result.WriteString(fmt.Sprintf("  Ошибка выполнения: %v\n", err))
		}
		text := decodeSFCOutput(raw)
		lines := strings.Split(text, "\n")
		for _, line := range lines {
			line = strings.TrimSpace(line)
			if line == "" {
				continue
			}
			result.WriteString("  " + line + "\n")
		}
	case "linux":
		result.WriteString(" Linux: аналог 'sfc' зависит от дистрибутива.\n")
		result.WriteString(" Debian/Ubuntu: выполните вручную 'sudo debsums -ca'\n")
		result.WriteString(" RHEL/Fedora: выполните вручную 'rpm -Va'\n")
	default:
		result.WriteString(" Платформа не поддерживается.\n")
	}

	return result.String()
}

func decodeSFCOutput(raw []byte) string {
	if len(raw) < 2 {
		return string(raw)
	}

	start := 0
	if raw[0] == 0xFF && raw[1] == 0xFE {
		start = 2
	}

	var u16 []uint16
	for i := start; i+1 < len(raw); i += 2 {
		u16 = append(u16, uint16(raw[i])|uint16(raw[i+1])<<8)
	}
	decoded := string(utf16.Decode(u16))

	// Нормализуем все переносы
	decoded = strings.ReplaceAll(decoded, "\r\n", "\n")
	decoded = strings.ReplaceAll(decoded, "\r", "\n")

	lines := strings.Split(decoded, "\n")
	var result []string
	var lastProgress string

	for _, line := range lines {
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}

		// Все строки прогресса — запоминаем только последнюю
		if strings.HasPrefix(line, "Проверка ") && strings.HasSuffix(line, "завершена.") {
			lastProgress = line
			continue
		}

		// Дедуплицируем остальные строки
		dup := false
		for _, existing := range result {
			if existing == line {
				dup = true
				break
			}
		}
		if !dup {
			result = append(result, line)
		}
	}

	// Вставляем последнюю строку прогресса перед финальным результатом
	if lastProgress != "" {
		insertAt := len(result)
		for i, l := range result {
			if strings.Contains(l, "Защита ресурсов") {
				insertAt = i
				break
			}
		}
		if insertAt <= len(result) {
			result = append(result[:insertAt], append([]string{lastProgress}, result[insertAt:]...)...)
		} else {
			result = append(result, lastProgress)
		}
	}

	return strings.Join(result, "\n")
}

// sysFile — чтение sysfs-файла без ошибок: пустая строка при недоступности.
func sysFile(path string) string {
	data, err := os.ReadFile(path)
	if err != nil {
		return ""
	}
	s := strings.TrimSpace(string(data))
	if s == "Unknown" || s == "Not Specified" || s == "Default string" || s == "To Be Filled By O.E.M." {
		return ""
	}
	return s
}

func orDash(s string) string {
	if s == "" {
		return "-"
	}
	return s
}

// getDNSInfo получает информацию о DNS‑серверах
func getDNSInfo() (string, error) {
	if isWindows() {
		cmd := exec.Command("ipconfig", "/all")
		output, err := cmd.Output()
		if err != nil {
			return "", err
	}

		lines := strings.Split(string(output), "\n")
		for _, line := range lines {
			if strings.Contains(line, "DNS Servers") || strings.Contains(line, "DNS‑серверы") {
				parts := strings.Split(line, ":")
				if len(parts) > 1 {
					dns := strings.TrimSpace(parts[1])

			}
		}
	}

		// Linux: читаем /etc/resolv.conf

		if err != nil {

	}

	}
	}

// ScanOpenPorts сканирует стандартные порты на localhost — показывает,
// какие службы принимают подключения на этой машине.
func ScanOpenPorts() string {
	var ports []int
	for p := range knownPorts {
		ports = append(ports, p)
	}
	sort.Ints(ports)

	type openPort struct {
		port  int
		state string
	}
	var (
		mu    sync.Mutex
		found []openPort
		wg    sync.WaitGroup
	)

	for _, p := range ports {
		wg.Add(1)
		go func(p int) {
			defer wg.Done()
			conn, err := net.DialTimeout("tcp", fmt.Sprintf("127.0.0.1:%d", p), 300*time.Millisecond)
			if err != nil {
				return
			}
			conn.Close()
			mu.Lock()
			found = append(found, openPort{port: p, state: knownPorts[p]})
			mu.Unlock()
		}(p)
	}
	wg.Wait()

	var b strings.Builder
	if len(found) == 0 {
		return "Открытые порты (localhost): слушающих портов из стандартного списка не обнаружено"
	}
	b.WriteString(fmt.Sprintf("Открытые порты (localhost): найдено %d из %d проверенных\n\n", len(found), len(ports)))
	b.WriteString("  Порт    Служба\n")
	for _, op := range found {
		b.WriteString(fmt.Sprintf("  %-8d%s\n", op.port, op.state))
	}
	return b.String()
}

// GetTopProcessesInfo — топ-10 процессов по накопленному CPU и по RAM.

	procs, err := process.Processes()
	if err != nil {
		return fmt.Sprintf("Процессы: ошибка получения списка: %v", err)
	}
	var rows []procRow
	for _, p := range procs {
		name, _ := p.Name()
		if name == "" {
			continue
		}
		t, err1 := p.Times()
		mi, err2 := p.MemoryInfo()
		if err1 != nil || err2 != nil {
			continue
		}
		rows = append(rows, procRow{name, p.Pid, t.Total(), float64(mi.RSS) / 1024 / 1024})

	}

	byCPU := append([]procRow(nil), rows...)
	sort.Slice(byCPU, func(i, j int) bool { return byCPU[i].cpu > byCPU[j].cpu })
	byRAM := append([]procRow(nil), rows...)
	sort.Slice(byRAM, func(i, j int) bool { return byRAM[i].mem > byRAM[j].mem })

	var b strings.Builder
	fmt.Fprintln(&b, "Топ-10 процессов по времени CPU:")
	for i := 0; i < len(byCPU) && i < 10; i++ {
		fmt.Fprintf(&b, "  %-25s PID %-7d CPU %.1f с\n", byCPU[i].name, byCPU[i].pid, byCPU[i].cpu)
	}
	fmt.Fprintln(&b, "Топ-10 процессов по RAM:")
	for i := 0; i < len(byRAM) && i < 10; i++ {

// runPS выполняет PowerShell-команду и возвращает вывод как строку.
// Кодировка принудительно UTF-8 — иначе русские названия процессов
// и служб на Windows превращаются в кракозябры.
func runPS(command string) (string, error) {
	full := `[Console]::OutputEncoding=[Text.Encoding]::UTF8; ` + command
	out, err := exec.Command("powershell", "-NoProfile", "-Command", full).Output()
	return string(out), err
}

// GetProcessesInfo получает информацию о запущенных процессах:
// топ-10 по CPU и топ-10 по памяти.
func GetProcessesInfo() string {
	var result strings.Builder

	if runtime.GOOS == "linux" {
		out, err := exec.Command("ps", "aux").Output()
		if err != nil {
			return "Процессы: не удалось получить информацию (ps aux)"
		}
		result.WriteString("Запущенные процессы (Linux, ps aux):\n")
		result.WriteString(string(out))
		return result.String()
	}

	if runtime.GOOS != "windows" {
		return "Процессы: не удалось получить информацию для текущей ОС"
	}

	// Топ-10 по CPU. Фильтр по $null нужен: у свежезапущенных процессов
	// счётчик CPU ещё не заполнен и сортировать их бессмысленно.
	psTop := `Get-Process | Where-Object { $_.CPU -ne $null } | ` +
		`Sort-Object CPU -Descending | Select-Object -First 10 ` +
		`@{n='Имя';e={$_.ProcessName}}, ` +
		`@{n='PID';e={$_.Id}}, ` +
		`@{n='CPU(сек)';e={[math]::Round($_.CPU,1)}}, ` +
		`@{n='RAM(МБ)';e={[math]::Round($_.WorkingSet64/1MB,0)}} | ` +
		`Format-Table -AutoSize | Out-String -Width 120`

	out, err := runPS(psTop)
	if err != nil {
		return "Процессы: не удалось получить информацию (PowerShell)"
	}
	result.WriteString("Процессы — топ-10 по CPU:\n")
	result.WriteString(strings.TrimSpace(out) + "\n")

	// Топ-10 по памяти
	psRam := `Get-Process | Sort-Object WorkingSet64 -Descending | ` +
		`Select-Object -First 10 ` +
		`@{n='Имя';e={$_.ProcessName}}, ` +
		`@{n='PID';e={$_.Id}}, ` +
		`@{n='RAM(МБ)';e={[math]::Round($_.WorkingSet64/1MB,0)}} | ` +
		`Format-Table -AutoSize | Out-String -Width 120`

	out, err = runPS(psRam)
	if err == nil {
		result.WriteString("\nПроцессы — топ-10 по памяти:\n")
		result.WriteString(strings.TrimSpace(out) + "\n")
	}

	return result.String()
}

// GetServicesInfo получает сводку по службам:
// общая статистика + службы с автозапуском, упавшие с ошибкой.
func GetServicesInfo() string {
	if runtime.GOOS != "windows" {
		return "Службы: доступно только на Windows"
	}

	var result strings.Builder

	// Общая статистика
	psStats := `$all = Get-Service; ` +
		`$running = ($all | Where-Object {$_.Status -eq 'Running'}).Count; ` +
		`$stopped = ($all | Where-Object {$_.Status -eq 'Stopped'}).Count; ` +
		`Write-Output ("Всего служб: " + $all.Count + " | Запущено: " + $running + " | Остановлено: " + $stopped)`

	out, err := runPS(psStats)
	if err == nil {
		result.WriteString("Службы — сводка:\n " + strings.TrimSpace(out) + "\n")
	} else {
		result.WriteString("Службы: не удалось получить статистику\n")
	}

	// Автозапуск, но не запущены с кодом ошибки — признак проблем.
	// Фильтр по ExitCode отсеивает службы, которые стоят легитимно
	// (отложенный старт, триггеры запуска).
	psAuto := `Get-WmiObject Win32_Service | ` +
		`Where-Object { $_.StartMode -eq 'Auto' -and $_.State -ne 'Running' -and $_.ExitCode -ne 0 } | ` +
		`Sort-Object DisplayName | Select-Object -First 15 ` +
		`@{n='Имя';e={$_.DisplayName}}, ` +
		`@{n='Состояние';e={$_.State}}, ` +
		`@{n='Код выхода';e={$_.ExitCode}} | ` +
		`Format-Table -AutoSize | Out-String -Width 120`

	out, err = runPS(psAuto)
	if err != nil {
		result.WriteString("Службы: не удалось получить список проблемных служб\n")
	} else if clean := strings.TrimSpace(out); clean == "" {
		result.WriteString("Службы с автозапуском, не запустившиеся с ошибкой: не обнаружены\n")
	} else {
		result.WriteString("Службы — автозапуск, но не запущены (с кодом ошибки):\n")
		result.WriteString(clean + "\n")
	}

	return result.String()
}

// GetUSBInfo — список USB-устройств (Linux: /sys/bus/usb/devices, резерв lsusb).
func GetUSBInfo() string {
	var b strings.Builder
	entries, _ := filepath.Glob("/sys/bus/usb/devices/*")
	count := 0
	for _, dir := range entries {
		base := filepath.Base(dir)
		if strings.Contains(base, ":") { // ':1.0' — интерфейсы, пропускаем
			continue
		}
		product, vendor, id := sysFile(dir+"/product"), sysFile(dir+"/idVendor"), sysFile(dir+"/idProduct")
		if product == "" && vendor == "" {
			continue
		}
		count++
		fmt.Fprintf(&b, "%d. %s (VID:PID %s:%s)\n", count, orDash(product), orDash(vendor), orDash(id))
	}
	if count == 0 {
		out, err := exec.Command("lsusb").Output()
		if err != nil {
			return "USB-устройства: получить список не удалось (lsusb недоступен)"
		}
		return "USB-устройства (lsusb):\n" + string(out)
	}
	return b.String()
}

// +build linux windows

package system

import (
	"fmt"
	"os/exec"
	"strings"
	"runtime"

	// Для WMI на Windows
	ole "github.com/go-ole/go-ole"
	oleutil "github.com/go-ole/go-ole/oleutil"
)

// GetUSBInfo получает информацию о подключённых USB‑устройствах
func GetUSBInfo() string {
	var result strings.Builder

	if runtime.GOOS == "linux" {
		cmd := exec.Command("lsusb")
		out, err := cmd.Output()
		if err != nil {
			return "USB‑устройства: не удалось получить информацию (команда lsusb недоступна или требуются права root)"
	}

		lines := strings.Split(string(out), "\n")
		result.WriteString("USB‑устройства (Linux):\n")

		for _, line := range lines {
			trimmed := strings.TrimSpace(line)
			if trimmed != "" {
				result.WriteString("  " + trimmed + "\n")
		}
	}

		if result.Len() <= len("USB‑устройства (Linux):\n") {
			return "USB‑устройства: не обнаружены"
	}
	return result.String()

	} else if runtime.GOOS == "windows" {
		// Сначала пробуем WMI
		devices, err := getUSBViaWMI()
		if err == nil && len(devices) > 0 {
			result.WriteString("USB‑устройства (WMI):\n")
			for _, device := range devices {
				result.WriteString("  " + device + "\n")
		}
		return result.String()
	}

		// Если WMI не сработал, пробуем PowerShell
		devices, err = getUSBViaPowerShell()
		if err == nil && len(devices) > 0 {
			result.WriteString("USB‑устройства (PowerShell):\n")
			for _, device := range devices {
				result.WriteString("  " + device + "\n")
		}
		return result.String()
	}
	}

	return "USB‑устройства: не удалось получить информацию"
}

// getUSBViaWMI получает USB‑устройства через WMI (Win32_USBHub)
func getUSBViaWMI() ([]string, error) {
	err := ole.CoInitialize(0)
	if err != nil {
		return nil, err
	}
	defer ole.CoUninitialize()

	unknown, err := oleutil.CreateObject("WbemScripting.SWbemLocator")
	if err != nil {
		return nil, err
	}
	defer unknown.Release()

	locator, err := unknown.QueryInterface(ole.IID_IDispatch)
	if err != nil {
		return nil, err
	}
	defer locator.Release()

	serviceRaw, err := oleutil.CallMethod(locator, "ConnectServer", ".", "root\\CIMV2")
	if err != nil {
		return nil, fmt.Errorf("не удалось подключиться к WMI: %v", err)
	}

	service := serviceRaw.ToIDispatch()
	defer service.Release()

	query := "SELECT Name, DeviceID FROM Win32_USBHub"
	resultsRaw, err := oleutil.CallMethod(service, "ExecQuery", query)
	if err != nil {
		return nil, fmt.Errorf("WMI запрос не выполнен: %v", err)
	}

	results := resultsRaw.ToIDispatch()
	defer results.Release()

	countVar, err := oleutil.GetProperty(results, "Count")
	if err != nil || countVar.Val == 0 {
		return nil, fmt.Errorf("USB‑устройства не найдены через WMI")
	}

	var devices []string
	for i := 0; i < int(countVar.Val); i++ {
		itemRaw, err := oleutil.CallMethod(results, "ItemIndex", i)
		if err != nil {
			continue
	}
	item := itemRaw.ToIDispatch()
	defer item.Release()

	nameRaw, err := oleutil.GetProperty(item, "Name")
	name := "Неизвестно"
	if err == nil {
		name = nameRaw.ToString()
	}

	deviceIDRaw, err := oleutil.GetProperty(item, "DeviceID")
	deviceID := "Неизвестно"
	if err == nil {
		deviceID = deviceIDRaw.ToString()
	}

	devices = append(devices, fmt.Sprintf("%s (ID: %s)", name, deviceID))
	}

	return devices, nil
}

// getUSBViaPowerShell получает USB‑устройства через PowerShell
func getUSBViaPowerShell() ([]string, error) {
	cmd := exec.Command("powershell", "-Command",
		"Get-PnpDevice -Class USB | Where-Object {$_.Present -eq $true} | Select-Object -ExpandProperty Name")
	output, err := cmd.Output()
	if err != nil {
		return nil, err
	}

	lines := strings.Split(strings.TrimSpace(string(output)), "\n")
	var devices []string

	for _, line := range lines {
		trimmed := strings.TrimSpace(line)
		if trimmed != "" && trimmed != "Name" {
			devices = append(devices, trimmed)
	}
	}

	if len(devices) == 0 {
		return nil, fmt.Errorf("USB‑устройства не найдены через PowerShell")
	}

	return devices, nil
}

package utils

import (

	"regexp"
	"runtime"
	"strconv"

// Ping выполняет пинг до хоста и возвращает среднее время отклика
func Ping(host string, count int) (string, error) {
	var cmd *exec.Cmd

	switch runtime.GOOS {
	case "windows":
		cmd = exec.Command("ping", host, "-n", strconv.Itoa(count))
	case "darwin", "linux":
		cmd = exec.Command("ping", host, "-c", strconv.Itoa(count))
	default:
		return "", fmt.Errorf("unsupported platform")
	}

	output, err := cmd.CombinedOutput()
	if err != nil {
		// Если пакеты потеряны, ping вернет ошибку выхода, но нам нужен текст
	}

	// Парсинг для Linux/macOS
	reUnix := regexp.MustCompile(`rtt [\w/]* = [\d.]+/([\d.]+)/[\d.]+/[\d.]+`)
	matchesUnix := reUnix.FindStringSubmatch(string(output))
	if len(matchesUnix) > 1 {
		return "Успех. Среднее время: " + matchesUnix[1] + " мс", nil
	}

	// Парсинг для Windows — поддерживает и русский, и английский вывод:
	//   Русский:    "Среднее = 26мс"  или  "Среднее = 26 мс"
	//   Английский: "Average = 26ms"  или  "Average = 26 ms"
	reWin := regexp.MustCompile(`(?:Среднее|Average)\s*=\s*(\d+)\s*м?ms`)
	matchesWin := reWin.FindStringSubmatch(string(output))
	if len(matchesWin) > 1 {
		return "Успех. Среднее время: " + matchesWin[1] + " мс", nil
	}

	// Если ничего не нашли, выводим сырой результат
	return "Не удалось распарсить результат:\n" + string(output), nil
}

func PrintSeparator() string {

// FormatBytes форматирует байты в читаемый вид (КБ, МБ, ГБ, ТБ)
func FormatBytes(bytes int64) string {
	const unit = 1024
	units := []string{"B", "KB", "MB", "GB", "TB"}

	if bytes < unit {
		return fmt.Sprintf("%d %s", bytes, units[0])
	}

	div, exp := int64(unit), 0
	for n := bytes / unit; n >= unit; n /= unit {
		div *= unit
		exp++
		if exp >= len(units)-1 { // защита от выхода за границы массива
			break
		}
	}

	// Ограничиваем максимальную единицу измерения (TB)
	if exp+1 >= len(units) {
		exp = len(units) - 2
	}

	return fmt.Sprintf("%.1f %s", float64(bytes)/float64(div), units[exp+1])
}
